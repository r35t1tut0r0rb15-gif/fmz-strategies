## Stop 0.7 % / target 0.5 % of the average price  [lines 100-136]
  100: stop_loss_pct = 0.7 // Risk 0.7% of capital
  101: take_profit_pct = 0.5 // Target 0.5% of capital
  102: 
  103: // Calculate EMAs
  104: ema_fast = ta.ema(close, length_fast)
  105: ema_slow = ta.ema(close, length_slow)
  106: 
  107: // Plot EMAs
  108: plot(ema_fast, color=color.blue, title="Fast EMA")
  109: plot(ema_slow, color=color.red, title="Slow EMA")
  110: 
  111: // Trading logic
  112: long_condition = ta.crossover(ema_fast, ema_slow)
  113: short_condition = ta.crossunder(ema_fast, ema_slow)
  114: 
  115: // Calculate stop loss and take profit levels
  116: stop_loss_long = strategy.position_avg_price * (1 - stop_loss_pct / 100)
  117: take_profit_long = strategy.position_avg_price * (1 + take_profit_pct / 100)
  118: 
  119: stop_loss_short = strategy.position_avg_price * (1 + stop_loss_pct / 100)
  120: take_profit_short = strategy.position_avg_price * (1 - take_profit_pct / 100)
  121: 
  122: // Enter and exit trades
  123: if (long_condition)
  124:     strategy.entry("Long", strategy.long)
  125: if (short_condition)
  126:     strategy.entry("Short", strategy.short)
  127: 
  128: // Exit long trades
  129: if (strategy.position_size > 0)
  130:     strategy.exit("Take Profit Long", "Long", limit=take_profit_long)
  131:     strategy.exit("Stop Loss Long", "Long", stop=stop_loss_long)
  132: 
  133: // Exit short trades
  134: if (strategy.position_size < 0)
  135:     strategy.exit("Take Profit Short", "Short", limit=take_profit_short)
  136:     strategy.exit("Stop Loss Short", "Short", stop=stop_loss_short)
