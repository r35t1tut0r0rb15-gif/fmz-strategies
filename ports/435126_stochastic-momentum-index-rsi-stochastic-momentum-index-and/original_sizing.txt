  163: //Fast RSI
  164: fastup = rma(max(change(close), 0), periodrsi)
  165: fastdown = rma(-min(change(close), 0), periodrsi)
  166: fastrsi = fastdown == 0 ? 100 : fastup == 0 ? 0 : 100 - (100 / (1 + fastup / fastdown))
  167: 
  168: //Stochastic Momentum Index
  169: ll = lowest (low, a)
  170: hh = highest (high, a)
  171: diff = hh - ll
  172: rdiff = close - (hh+ll)/2
  173: //avgrel = ema(ema(rdiff,b),b)
  174: //avgdiff = ema(ema(diff,b),b)
  175: avgrel = sma(sma(rdiff,b),b)
  176: avgdiff = sma(sma(diff,b),b)
  177: SMI = avgdiff != 0 ? (avgrel/(avgdiff/2)*100) : 0
  178: SMIsignal = ema(SMI,b)
  179: 
  180: //Lines
  181: plot(SMI, color = blue, linewidth = 3, title = "Stochastic Momentum Index")
  182: plot(SMIsignal, color = red, linewidth = 3, title = "SMI Signal Line")
  183: plot(limitsmi, color = black, title = "Over Bought")
  184: plot(-1 * limitsmi, color = black, title = "Over Sold")
  185: plot(0, color = blue, title = "Zero Line")

  187: //Color-Filter
  188: gb = close > open or usecol == false
  189: rb = close < open or usecol == false
  190: 
  191: //Body Filter
  192: nbody = abs(close - open)
  193: abody = sma(nbody, 10)
  194: body = nbody > abody / 3 or usebod == false
  195: 
  196: //Signals

  198: dn1 = SMI > limitsmi and gb and body and usesmi
  199: up2 = fastrsi < limitrsi and rb and body and usersi
  200: dn2 = fastrsi > 100 - limitrsi and gb and body and usersi
  201: exit = ((strategy.position_size > 0 and close > open) or (strategy.position_size < 0 and close < open)) and body
  202: 
  203: //Background
  204: redb = (SMI > limitsmi and usesmi) or (fastrsi > 100 - limitrsi and usersi)
  205: limeb = (SMI < -1 * limitsmi and usesmi) or (fastrsi < limitrsi and usersi)
  206: col = showbg == false ? na : redb ? red : limeb ? lime : na
  207: bgcolor(col, transp = 50)
  208: 
  209: //Trading
  210: profit = exit ? ((strategy.position_size > 0 and close > strategy.position_avg_price) or (strategy.position_size < 0 and close < strategy.position_avg_price)) ? 1 : -1 : profit[1]
  211: mult = usemar ? exit ? profit == -1 ? mult[1] * 2 : 1 : mult[1] : 1
  212: lot = strategy.position_size == 0 ? strategy.equity / close * capital / 100 * mult : lot[1]
