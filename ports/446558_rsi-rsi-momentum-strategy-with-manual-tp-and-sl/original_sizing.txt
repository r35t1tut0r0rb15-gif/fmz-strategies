## Take profit / stop loss percent of the average price  [lines 127-136]
  127: tp_percentage = input.float(1, title="Take Profit (%)")
  128: sl_percentage = input.float(1, title="Stop Loss (%)")
  129: 
  130: long_tp = strategy.position_avg_price * (1 + tp_percentage / 100)
  131: long_sl = strategy.position_avg_price * (1 - sl_percentage / 100)
  132: short_tp = strategy.position_avg_price * (1 - tp_percentage / 100)
  133: short_sl = strategy.position_avg_price * (1 + sl_percentage / 100)
  134: 
  135: strategy.exit("TP/SL Long", "RsiLE", limit=long_tp, stop=long_sl)
  136: strategy.exit("TP/SL Short", "RsiSE", limit=short_tp, stop=short_sl)
