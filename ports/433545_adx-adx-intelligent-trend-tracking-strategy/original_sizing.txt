  141: profitTargetMultiple = input(1.0, title="Profit Target Box Width Multiple", group = "Take Profit and Stop Loss")

  146: // When the ADX drops below threshold limit, then we consider the pair in consolidation. 
  147: // Set Box around highs and lows of the last 20 candles. with upper and lower boundaries. 
  148: // When price breaks outside of box, a trade is taken. (on close or on touch?)
  149: // Stop is placed, default 50%, of the size of the box. So if box is 200 pips, stop is at 100 pips. 
  150: // Profit target is 100% of the size of the box. Default. User can set a profit target of 0.5, 1 full size, 2 or 3. 
  151: 
  152: 
  153: dirmov(len) =>
  154: 	up = ta.change(high)
  155: 	down = -ta.change(low)
  156: 	truerange = ta.rma(ta.tr, len)
  157: 	plus = fixnan(100 * ta.rma(up > down and up > 0 ? up : 0, len) / truerange)
  158: 	minus = fixnan(100 * ta.rma(down > up and down > 0 ? down : 0, len) / truerange)
  159: 	[plus, minus]
  160: 
  161: adx(dilen, adxlen) => 
  162: 	[plus, minus] = dirmov(dilen)
  163: 	sum = plus + minus
  164: 	adx = 100 * ta.rma(math.abs(plus - minus) / (sum == 0 ? 1 : sum), adxlen)
  165: 
  166: adxHigh(dilen, adxlen) => 
  167: 	[plus, minus] = dirmov(dilen)
  168: 	plus
  169: 	
  170: adxLow(dilen, adxlen) => 
  171: 	[plus, minus] = dirmov(dilen)
  172: 	minus

  174: sig = adx(adxSmoothPeriod, adxPeriod)
