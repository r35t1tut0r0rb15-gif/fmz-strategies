## Stop 1.5 ATR / target 3 ATR from the signal close  [lines 101-168]
  101: atrLength = input.int(14, title="ATR Length", minval=1)
  102: atrMultiplierSL = input.float(1.5, title="Stop-Loss Multiplier (ATR)", minval=0.1)   // Stop-loss at 1.5x ATR
  103: atrMultiplierTP = input.float(3.0, title="Take-Profit Multiplier (ATR)", minval=0.1)   // Take-profit at 3x ATR
  104: pullbackRangePercent = input.float(1.0, title="Pullback Range (%)", minval=0.1) // 1% range for pullback around 20 EMA
  105: lengthKC = input.int(20, title="Length for Keltner Channels (Momentum Calculation)", minval=1)
  106: 
  107: // EMA settings
  108: ema20 = ta.ema(close, 20)
  109: ema50 = ta.ema(close, 50)
  110: ema100 = ta.ema(close, 100)
  111: ema200 = ta.ema(close, 200)
  112: 
  113: // ATR calculation
  114: atrValue = ta.atr(atrLength)
  115: 
  116: // Custom Momentum Calculation based on Linear Regression for Daily Timeframe
  117: highestHighKC = ta.highest(high, lengthKC)
  118: lowestLowKC = ta.lowest(low, lengthKC)
  119: smaCloseKC = ta.sma(close, lengthKC)
  120: 
  121: // Manually calculate the average of highest high and lowest low
  122: averageKC = (highestHighKC + lowestLowKC) / 2
  123: 
  124: // Calculate daily momentum using linear regression
  125: dailyMomentum = ta.linreg(close - (averageKC + smaCloseKC) / 2, lengthKC, 0) // Custom daily momentum calculation
  126: 
  127: // Fetch weekly data for momentum calculation using request.security()
  128: [weeklyHigh, weeklyLow, weeklyClose] = request.security(syminfo.tickerid, "W", [high, low, close])
  129: 
  130: // Calculate weekly momentum using linear regression on weekly timeframe
  131: weeklyHighestHighKC = ta.highest(weeklyHigh, lengthKC)
  132: weeklyLowestLowKC = ta.lowest(weeklyLow, lengthKC)
  133: weeklySmaCloseKC = ta.sma(weeklyClose, lengthKC)
  134: weeklyAverageKC = (weeklyHighestHighKC + weeklyLowestLowKC) / 2
  135: 
  136: weeklyMomentum = ta.linreg(weeklyClose - (weeklyAverageKC + weeklySmaCloseKC) / 2, lengthKC, 0) // Custom weekly momentum calculation
  137: 
  138: // EMA alignment condition (20 EMA > 50 EMA > 100 EMA > 200 EMA)
  139: emaAligned = ema20 > ema50 and ema50 > ema100 and ema100 > ema200
  140: 
  141: // Momentum increasing condition (daily and weekly momentum is positive and increasing)
  142: dailyMomentumIncreasing = dailyMomentum > 0 and dailyMomentum > dailyMomentum[1] //and dailyMomentum[1] > dailyMomentum[2]
  143: weeklyMomentumIncreasing = weeklyMomentum > 0 and weeklyMomentum > weeklyMomentum[1] //and weeklyMomentum[1] > weeklyMomentum[2]
  144: 
  145: // Redefine Pullback condition: price within 1% range of the 20 EMA
  146: upperPullbackRange = ema20 * (1 + pullbackRangePercent / 100)
  147: lowerPullbackRange = ema20 * (1 - pullbackRangePercent / 100)
  148: pullbackToEma20 = (close <= upperPullbackRange) and (close >= lowerPullbackRange)
  149: 
  150: // Entry condition: EMA alignment and momentum increasing on both daily and weekly timeframes
  151: longCondition = emaAligned and dailyMomentumIncreasing and weeklyMomentumIncreasing and pullbackToEma20
  152: 
  153: // Initialize stop loss and take profit levels as float variables
  154: var float longStopLevel = na
  155: var float longTakeProfitLevel = na
  156: 
  157: // Calculate stop loss and take profit levels based on ATR
  158: if (longCondition)
  159:     longStopLevel := close - (atrMultiplierSL * atrValue)  // Stop loss at 1.5x ATR below the entry price
  160:     longTakeProfitLevel := close + (atrMultiplierTP * atrValue) // Take profit at 3x ATR above the entry price
  161: 
  162: // Strategy execution
  163: if (longCondition)
  164:     strategy.entry("Long", strategy.long)
  165: 
  166: // Exit conditions: Stop-loss at 1.5x ATR and take-profit at 3x ATR
  167: if (strategy.position_size > 0)
  168:     strategy.exit("Take Profit/Stop Loss", "Long", stop=longStopLevel, limit=longTakeProfitLevel)
