  230:                   diffMA < 0 and _ma > _maRef ? color.maroon :
  231:                   diffMA <= 0 and _ma < _maRef ? color.red :

  278: short_stop_loss = entry_price * (1 + short_stop_loss_perc)     // +1%

  280: // Backtesting Orders
