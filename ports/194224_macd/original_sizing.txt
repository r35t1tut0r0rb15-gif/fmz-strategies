## Order helpers: sliced buy with timeouts, sell loop, rounding  [lines 31-141]
   31: Fixed = function(v) {
   32:     return Math.floor(v*1000)/1000;
   33: };

   35: // for orders
   36: WaitOrder = function(exchange, orderId, timeoutToCancel) {
   37:     var ts = (new Date()).getTime();
   38:     while (true) {
   39:         Sleep(3000);
   40:         var orderInfo = exchange.GetOrder(orderId);
   41:         if (!orderInfo) {
   42:             continue;
   43:         }
   44:         if (orderInfo.Status == ORDER_STATE_CLOSED || orderInfo.Status == ORDER_STATE_CANCELED) {
   45:             return orderInfo;
   46:         }
   47:         if (((new Date()).getTime() - ts) > timeoutToCancel) {
   48:             exchange.CancelOrder(orderId);
   49:         }
   50:     }
   51: };

   53: Buy = function(exchange, maxPrice, slidePrice, balanceRatio, timeoutS) {
   54:     var ts = (new Date()).getTime();
   55:     var account;
   56:     var dealAmount = 0.0;
   57:     var usedBlance = 0.0;
   58:     var maxBalanceUse = 0.0;
   59:     var isFirst = true;
   60:     do {
   61:         if (isFirst) {
   62:             isFirst = false;
   63:         } else {
   64:             Sleep(3000);
   65:         }
   66: 
   67:         var ticker = exchange.GetTicker();
   68:         if (!ticker) {
   69:             continue;
   70:         }
   71: 
   72:         var buyPrice = ticker.Sell + slidePrice;
   73: 
   74:         // Price too high, wait...
   75:         if (buyPrice > maxPrice) {
   76:             continue;
   77:         }
   78: 
   79:         // Initialize at first
   80:         if (!account) {
   81:             account = exchange.GetAccount();
   82:             if (!account) {
   83:                 continue;
   84:             }
   85:             // Initialize maxBalanceUse
   86:             maxBalanceUse = account.Balance * balanceRatio;
   87:         }
   88: 
   89:         var buyAmount = Fixed((maxBalanceUse - usedBlance) / buyPrice);
   90:         if (buyAmount < MinStock) {
   91:             break;
   92:         }
   93: 
   94:         orderId = exchange.Buy(buyPrice, buyAmount);
   95:         if (!orderId) {
   96:             Log(buyPrice, buyAmount, maxBalanceUse, usedBlance);
   97:             continue;
   98:         }
   99: 
  100:         var orderInfo = WaitOrder(exchange, orderId, timeoutS);
  101:         dealAmount += orderInfo.DealAmount;
  102:         usedBlance += orderInfo.Price * orderInfo.DealAmount;
  103:         if (orderInfo.Status == ORDER_STATE_CLOSED) {
  104:             break;
  105:         }
  106:     } while (((new Date()).getTime() - ts) < timeoutS);
  107: 
  108:     return {amount: dealAmount, price: (dealAmount > 0 ? usedBlance / dealAmount : 0)};
  109: };
  110: 
  111: Sell = function(exchange, sellAmount, slidePrice) {
  112:     // Account info must set
  113:     var account = exchange.GetAccount();
  114:     while (!account) {
  115:         Sleep(2000);
  116:         account = exchange.GetAccount();
  117:     }
  118: 
  119:     sellAmount = Math.min(sellAmount, account.Stocks);
  120: 
  121:     var cash = 0.0;
  122:     var remain = sellAmount;
  123: 
  124:     while (remain >= exchange.GetMinStock()) {
  125:         var ticker = exchange.GetTicker();
  126:         if (!ticker) {
  127:             Sleep(2000);
  128:             continue;
  129:         }
  130:         var sellPrice = ticker.Buy - slidePrice;
  131:         var sellOrderId = exchange.Sell(sellPrice, remain);
  132:         if (!sellOrderId) {
  133:             Sleep(2000);
  134:             continue;
  135:         }
  136:         var orderInfo = WaitOrder(exchange, sellOrderId, 10000);
  137:         remain -= orderInfo.DealAmount;
  138:         cash += orderInfo.Price * orderInfo.DealAmount;
  139:     }
  140:     return {amount: sellAmount, price: (sellAmount > 0 ? cash / sellAmount : 0)};
  141: };
