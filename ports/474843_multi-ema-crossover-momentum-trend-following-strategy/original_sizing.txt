## strategy(): 10 % of equity; target 10 % / stop 5 % of the current close  [lines 87-140]
   87: strategy("Pavan Guppy Strategy", shorttitle="Pavan Avg", overlay=true, 
   88:          default_qty_type=strategy.percent_of_equity, default_qty_value=10)
   89: 
   90: // Short-term EMAs
   91: shortEMA1 = ta.ema(close, 3)
   92: shortEMA2 = ta.ema(close, 5)
   93: shortEMA3 = ta.ema(close, 8)
   94: shortEMA4 = ta.ema(close, 10)
   95: shortEMA5 = ta.ema(close, 12)
   96: shortEMA6 = ta.ema(close, 15)
   97: 
   98: // Long-term EMAs
   99: longEMA1 = ta.ema(close, 30)
  100: longEMA2 = ta.ema(close, 35)
  101: longEMA3 = ta.ema(close, 40)
  102: longEMA4 = ta.ema(close, 45)
  103: longEMA5 = ta.ema(close, 50)
  104: longEMA6 = ta.ema(close, 60)
  105: 
  106: // Average short-term EMAs
  107: shortAvg = (shortEMA1 + shortEMA2 + shortEMA3 + shortEMA4 + shortEMA5 + shortEMA6) / 6.0
  108: 
  109: // Average long-term EMAs
  110: longAvg = (longEMA1 + longEMA2 + longEMA3 + longEMA4 + longEMA5 + longEMA6) / 6.0
  111: 
  112: // Plot averaged EMAs
  113: plot(shortAvg, color=color.green, linewidth=2, title="Averaged Short-term EMAs")
  114: plot(longAvg, color=color.red, linewidth=2, title="Averaged Long-term EMAs")
  115: 
  116: // Define the target and stop loss percentages
  117: takeProfitPerc = 10
  118: stopLossPerc = 5
  119: 
  120: // Generate buy signal when shortAvg crosses above longAvg
  121: if ta.crossover(shortAvg, longAvg)
  122:     strategy.entry("Buy", strategy.long)
  123: 
  124: // Generate sell signal when shortAvg crosses below longAvg
  125: if ta.crossunder(shortAvg, longAvg)
  126:     strategy.entry("Sell", strategy.short)
  127: 
  128: // Calculate take profit and stop loss prices for long trades
  129: longTakeProfit = close * (1 + (takeProfitPerc / 100.0))
  130: longStopLoss = close * (1 - (stopLossPerc / 100.0))
  131: 
  132: // Set take profit and stop loss for long positions
  133: strategy.exit("Take Profit/Stop Loss", from_entry="Buy", limit=longTakeProfit, stop=longStopLoss)
  134: 
  135: // Calculate take profit and stop loss prices for short trades
  136: shortTakeProfit = close * (1 - takeProfitPerc / 100.0)
  137: shortStopLoss = close * (1 + stopLossPerc / 100.0)
  138: 
  139: // Set take profit and stop loss for short positions
  140: strategy.exit("Take Profit/Stop Loss", from_entry="Sell", limit=shortTakeProfit, stop=shortStopLoss)
