## Risk 2 % of equity over a 1 ATR stop; target 0.75 ATR  [lines 133-173]
  133: rrRatio = input.float(0.75, title="Risk/Reward Ratio", step=0.05)  // Lower RRR to achieve a high win rate
  134: riskPerTrade = input.float(2.0, title="Risk per Trade (%)", step=0.1) / 100  // 2% risk per trade
  135: 
  136: // Bollinger Bands Calculation
  137: basis = ta.sma(close, bbLength)
  138: dev = bbMult * ta.stdev(close, bbLength)
  139: upperBand = basis + dev
  140: lowerBand = basis - dev
  141: 
  142: // RSI Calculation
  143: rsi = ta.rsi(close, rsiLength)
  144: 
  145: // ATR Calculation for Stop Loss
  146: atr = ta.atr(atrLength)
  147: 
  148: // Entry Conditions: Mean Reversion
  149: longCondition = close < lowerBand and rsi < rsiOversold
  150: shortCondition = close > upperBand and rsi > rsiOverbought
  151: 
  152: // Stop Loss and Take Profit based on ATR
  153: longStopLoss = close - atr * 1.0  // 1x ATR stop loss for long trades
  154: shortStopLoss = close + atr * 1.0  // 1x ATR stop loss for short trades
  155: 
  156: longTakeProfit = close + (close - longStopLoss) * rrRatio  // 0.75x ATR take profit
  157: shortTakeProfit = close - (shortStopLoss - close) * rrRatio  // 0.75x ATR take profit
  158: 
  159: // Calculate position size based on risk
  160: equity = strategy.equity
  161: riskAmount = equity * riskPerTrade
  162: qtyLong = riskAmount / (close - longStopLoss)
  163: qtyShort = riskAmount / (shortStopLoss - close)
  164: 
  165: // Long Trade
  166: if (longCondition)
  167:     strategy.entry("Long", strategy.long, qty=qtyLong)
  168:     strategy.exit("Take Profit/Stop Loss", from_entry="Long", limit=longTakeProfit, stop=longStopLoss)
  169: 
  170: // Short Trade
  171: if (shortCondition)
  172:     strategy.entry("Short", strategy.short, qty=qtyShort)
  173:     strategy.exit("Take Profit/Stop Loss", from_entry="Short", limit=shortTakeProfit, stop=shortStopLoss)
