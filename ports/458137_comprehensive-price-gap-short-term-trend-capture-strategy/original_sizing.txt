## Gap threshold 150 / target 50 points (price units)  [lines 308-342]
  308: targetPoints = input.int(50, title="Target Points", minval=1)
  309: gapThreshold = input.int(150, title="Gap Threshold (in points)", minval=0)

  311: // Calculate gap
