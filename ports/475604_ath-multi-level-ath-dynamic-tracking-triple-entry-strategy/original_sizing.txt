  120: // === Update All-Time High ===
  121: if na(all_time_high)
  122:     all_time_high := high
  123: else
  124:     // Update ATH to the previous bar's high to exclude current bar's high
  125:     all_time_high := math.max(all_time_high[1], high[1])

  127:         // New ATH reached on the previous bar

  129:         dip2_bought := false
  130:         dip3_bought := false

  132: // === Calculate Percentage Drop from ATH ===
  133: percent_drop = (all_time_high - close) / all_time_high * 100.0

  135: // === Define Dip Conditions ===

  137: buyDip2 = (percent_drop >= dip2) and not dip2_bought
  138: buyDip3 = (percent_drop >= dip3) and not dip3_bought

  140: // === Calculate Quantity to Buy ===

  142: 
  143: // === Execute Buys on Dips ===
