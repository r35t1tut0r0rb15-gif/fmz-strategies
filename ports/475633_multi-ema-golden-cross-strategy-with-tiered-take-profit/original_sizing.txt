## qty 0.5 per id; stop / TP1 / TP2 from the 20-bar low  [lines 103-144]
  103: lookback_bars = input.int(20, title="Lookback bars for lowest low")
  104: pip_buffer = input.float(0.0003, title="Stop-loss buffer (pips)")  // Fixed default pip value (e.g., 3 pips for 5-digit pairs)
  105: tp_multiplier1 = input.float(1.0, title="Take-profit multiplier 1")
  106: tp_multiplier2 = input.float(1.5, title="Take-profit multiplier 2")
  107: 
  108: // Calculate EMAs
  109: ema25 = ta.ema(close, ema_short_length)
  110: ema50 = ta.ema(close, ema_mid_length)
  111: ema100 = ta.ema(close, ema_long_length)
  112: 
  113: // Golden Cross condition (EMA25 > EMA50 > EMA100)
  114: golden_cross = ema25 > ema50 and ema50 > ema100
  115: 
  116: // Entry condition: Candle crosses above EMA25 after a golden cross
  117: cross_above_ema25 = ta.crossover(close, ema25)
  118: entry_condition = golden_cross and cross_above_ema25
  119: 
  120: // Stop-loss and take-profit calculation
  121: lowest_low = ta.lowest(low, lookback_bars)
  122: var float entry_price = na
  123: var float stop_loss = na
  124: var float take_profit1 = na
  125: var float take_profit2 = na
  126: 
  127: if (entry_condition)
  128:     entry_price := close
  129:     stop_loss := lowest_low - pip_buffer
  130:     take_profit1 := entry_price + (entry_price - stop_loss) * tp_multiplier1
  131:     take_profit2 := entry_price + (entry_price - stop_loss) * tp_multiplier2
  132:     strategy.entry("Buy1", strategy.long, qty=0.5)  // First 50%
  133:     strategy.entry("Buy2", strategy.long, qty=0.5)  // Second 50%
  134: 
  135: // Separate exit conditions for each entry
  136: cross_below_ema100 = ta.crossunder(close, ema100)
  137: exit_condition1 = close >= take_profit1
  138: exit_condition2 = close >= take_profit2
  139: exit_condition_sl = close <= stop_loss
  140: 
  141: if (exit_condition1 or cross_below_ema100)
  142:     strategy.close("Buy1")
  143: if (exit_condition2 or cross_below_ema100 or exit_condition_sl)
  144:     strategy.close("Buy2")
