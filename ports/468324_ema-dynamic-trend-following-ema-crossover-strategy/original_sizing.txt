## Target 55 / stop 30 ticks; trail_offset 25 x mintick  [lines 169-217]

  193: // Determine trend direction
  194: bullishTrend = ema55 > ema200
  195: bearishTrend = ema55 < ema200

  197: // Define entry conditions
  198: longCondition = bullishTrend and ta.crossover(closePrice, lowCustomPeriod) and ta.crossover(closePrice, ema55)
  199: shortCondition = bearishTrend and ta.crossunder(closePrice, highCustomPeriod) and ta.crossunder(closePrice, ema55)
  200: 
  201: // Entry conditions and auto take profit, stop loss, and trailing stop
  202: if (boxClose)
  203:     if (longCondition)
  204:         takeProfitPriceLong = closePrice + takeProfitTicks * syminfo.mintick
  205:         stopLossPriceLong = closePrice - stopLossTicks * syminfo.mintick

  207:         strategy.exit("Take Profit Long", "Long", limit=takeProfitPriceLong, stop=stopLossPriceLong, trail_offset=trailingStopTicks * syminfo.mintick)
  208:         // Plot visual signal for long entry
  209:         label.new(bar_index, closePrice, "Buy", color=color.green, textcolor=color.white, style=label.style_label_up, size=size.small)
  210:         // Send alert for long entry
  211:         alert("Long entry signal - price: " + str.tostring(closePrice), alert.freq_once_per_bar)
  212:         
  213:     if (shortCondition)
  214:         takeProfitPriceShort = closePrice - takeProfitTicks * syminfo.mintick
  215:         stopLossPriceShort = closePrice + stopLossTicks * syminfo.mintick
  216:         strategy.entry("Short", strategy.short)
  217:         strategy.exit("Take Profit Short", "Short", limit=takeProfitPriceShort, stop=stopLossPriceShort, trail_offset=trailingStopTicks * syminfo.mintick)
