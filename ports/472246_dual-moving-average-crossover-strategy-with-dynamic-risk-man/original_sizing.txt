## Stop 1 % / target 2 % of the signal close; capital 10000  [lines 94-145]
   94: strategy("BTCUSD Daily Strategy - Market Orders Only", overlay=true, initial_capital=10000, currency=currency.USD)
   95: 
   96: // Configurable Inputs
   97: stop_loss_percent = input.float(title="Stop Loss (%)", defval=1.0, minval=0.0, step=0.1)
   98: take_profit_percent = input.float(title="Take Profit (%)", defval=2.0, minval=0.0, step=0.1)
   99: short_ma_length = input.int(title="Short MA Length", defval=9, minval=1)
  100: long_ma_length = input.int(title="Long MA Length", defval=21, minval=1)
  101: 
  102: // Moving Averages
  103: short_ma = ta.sma(close, short_ma_length)
  104: long_ma = ta.sma(close, long_ma_length)
  105: 
  106: // Plotting Moving Averages
  107: plot(short_ma, color=color.blue, title="Short MA")
  108: plot(long_ma, color=color.red, title="Long MA")
  109: 
  110: // Buy and Sell Signals
  111: buy_signal = ta.crossover(short_ma, long_ma)
  112: sell_signal = ta.crossunder(short_ma, long_ma)
  113: 
  114: // Market Buy Logic
  115: if (buy_signal and strategy.position_size <= 0)
  116:     // Close any existing short position
  117:     if (strategy.position_size < 0)
  118:         strategy.close(id="Market Sell")
  119:     
  120:     // Calculate Stop Loss and Take Profit Prices
  121:     entry_price = close
  122:     long_stop = entry_price * (1 - stop_loss_percent / 100)
  123:     long_take_profit = entry_price * (1 + take_profit_percent / 100)
  124: 
  125:     // Enter Long Position
  126:     strategy.entry(id="Market Buy", direction=strategy.long)
  127:     strategy.exit(id="Exit Long", from_entry="Market Buy", stop=long_stop, limit=long_take_profit)
  128: 
  129:     // Alert for Market Buy
  130:     alert("Market Buy Signal at price " + str.tostring(close) + ". Stop Loss: " + str.tostring(long_stop) + ", Take Profit: " + str.tostring(long_take_profit), alert.freq_once_per_bar_close)
  131: 
  132: // Market Sell Logic
  133: if (sell_signal and strategy.position_size >= 0)
  134:     // Close any existing long position
  135:     if (strategy.position_size > 0)
  136:         strategy.close(id="Market Buy")
  137: 
  138:     // Calculate Stop Loss and Take Profit Prices
  139:     entry_price = close
  140:     short_stop = entry_price * (1 + stop_loss_percent / 100)
  141:     short_take_profit = entry_price * (1 - take_profit_percent / 100)
  142: 
  143:     // Enter Short Position
  144:     strategy.entry(id="Market Sell", direction=strategy.short)
  145:     strategy.exit(id="Exit Short", from_entry="Market Sell", stop=short_stop, limit=short_take_profit)
