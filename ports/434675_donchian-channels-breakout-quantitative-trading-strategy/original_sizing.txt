## Money-per-trade quantity and the monetary stop  [lines 167-199]
  167: instrument = input(1, title = "Select 1: Stock/Forex, 2: Future")
  168: money = input(10000, title = "Money for each trade")
  169: backtest_start = input(2000, "Insert first year to backtest")
  170: period = input(50, title = "Period in bars of Donchian Channel")
  171: monetary_stoploss = input(1000, title = "Monetary Stop Loss")
  172: 
  173: quantity = if instrument != 1 
  174:     1
  175: else
  176:     int(money / close)
  177:     
  178: upBarrier = highest(high,period)
  179: downBarrier = lowest(low,period)
  180: up = highest(high,period / 4)
  181: down = lowest(low,period / 4)
  182: 
  183: plot(upBarrier, color=color.green, linewidth=2)
  184: plot(downBarrier, color=color.red, linewidth=2)
  185: plot(up, color=color.lime, linewidth=1)
  186: plot(down, color=color.orange, linewidth=2)
  187: 
  188: longCondition = crossover(close, upBarrier[1]) and year >= backtest_start
  189: 
  190: if (longCondition)
  191:     strategy.entry("Long", strategy.long, quantity, when = strategy.position_size == 0)
  192: 
  193: closeCondition = crossunder(close, down[1]) or down < down[1]
  194: 
  195: if (closeCondition)
  196:     strategy.close("Long", comment = "Trailing")
  197:     
  198: stop_level = strategy.position_avg_price - monetary_stoploss / strategy.position_size
  199: strategy.exit("StopLoss", from_entry = "Long", stop = stop_level)
