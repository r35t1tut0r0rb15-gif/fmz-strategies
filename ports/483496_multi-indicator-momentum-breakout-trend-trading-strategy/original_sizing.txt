  129: stopLossPerc = input(2, title="Stop Loss %") / 100
  130: takeProfitPerc = stopLossPerc * riskRewardRatio

## Tick bracket from position_avg_price  [lines 132-139]

  135: shortSL = strategy.position_avg_price * (1 + stopLossPerc)
  136: shortTP = strategy.position_avg_price * (1 - takeProfitPerc)

  138: strategy.exit("Take Profit / Stop Loss", from_entry="Long", loss=longSL, profit=longTP)
