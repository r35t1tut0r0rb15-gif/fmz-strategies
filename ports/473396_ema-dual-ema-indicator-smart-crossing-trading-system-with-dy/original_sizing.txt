  106: // 格式化时间显示 (UTC+8)

  123: 
  124: if crossUnder
  125:     if strategy.position_size > 0  // 如果持有多仓
  126:         strategy.close("做多")     // 先平掉多仓
  127:     strategy.entry("做空", strategy.short)  // 开空仓

  135: if strategy.position_size < 0  // 空仓止损止盈
