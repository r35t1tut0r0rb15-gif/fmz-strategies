## qty 20 / close (20 % of equity default); stop 1 % / target 2 %  [lines 107-145]
  107: strategy("Estrategia Intradía - Cruce EMA + RSI - Optimizado", overlay=true, pyramiding=0, default_qty_type=strategy.percent_of_equity, default_qty_value=20)
  108: 
  109: // Parámetros CON rangos de optimización
  110: ema_fast_length = input.int(title="Período EMA Rápida", defval=12, minval=5, maxval=30, step=1)
  111: ema_slow_length = input.int(title="Período EMA Lenta", defval=26, minval=15, maxval=50, step=1)
  112: rsi_length = input.int(title="Período RSI", defval=14, minval=7, maxval=21, step=1)
  113: rsi_overbought = input.int(title="Nivel de Sobrecompra RSI", defval=70, minval=60, maxval=80, step=1)
  114: rsi_oversold = input.int(title="Nivel de Sobreventa RSI", defval=30, minval=20, maxval=40, step=1)
  115: stop_loss_percent = input.float(title="Stop Loss (%)", defval=1.0, minval=0.1, maxval=3.0, step=0.1)
  116: take_profit_percent = input.float(title="Take Profit (%)", defval=2.0, minval=0.5, maxval=5.0, step=0.1)
  117: 
  118: // Cálculos
  119: ema_fast = ta.ema(close, ema_fast_length)
  120: ema_slow = ta.ema(close, ema_slow_length)
  121: rsi = ta.rsi(close, rsi_length)
  122: 
  123: // Condiciones de entrada
  124: longCondition = ta.crossover(ema_fast, ema_slow) and rsi > 50
  125: shortCondition = ta.crossunder(ema_fast, ema_slow) and rsi < 50
  126: 
  127: // Gestión de entradas y salidas
  128: var float longQty = na
  129: var float shortQty = na
  130: 
  131: if longCondition
  132:     longQty := 20 / close
  133:     strategy.entry("Long", strategy.long, qty=longQty)
  134:     if stop_loss_percent > 0 and take_profit_percent > 0
  135:         strategy.exit("Exit Long", "Long", stop=close * (1 - stop_loss_percent / 100), limit=close * (1 + take_profit_percent / 100))
  136: 
  137: if strategy.position_size > 0 and ta.crossunder(ema_fast, ema_slow)
  138:     strategy.close("Long")
  139:     longQty := na
  140: 
  141: if shortCondition
  142:     shortQty := 20 / close
  143:     strategy.entry("Short", strategy.short, qty=shortQty)
  144:     if stop_loss_percent > 0 and take_profit_percent > 0
  145:         strategy.exit("Exit Short", "Short", stop=close * (1 + stop_loss_percent / 100), limit=close * (1 - take_profit_percent / 100))
