## Stop 1.5 ATR / target 2 ATR from the signal close  [lines 115-145]
  115: atrPeriod = input.int(14, title="ATR Period")
  116: stopLossMultiplier = input.float(1.5, minval=0.0, maxval = 20.0, step = 0.1 ,title="Stop Loss Multiplier")
  117: takeProfitMultiplier = input.float(2.0, minval=0.0 , maxval = 20.0, step = 0.1,title="Take Profit Multiplier")
  118: 
  119: // --- Trendilo ---
  120: trendiloPeriod = input.int(50, title="Trendilo Period")
  121: trendilo = ta.sma(close, trendiloPeriod)
  122: 
  123: // --- MACD ---
  124: [macdLine, signalLine, _] = ta.macd(close, 12, 26, 9)
  125: macdZeroCrossUp = ta.crossover(macdLine, 0)
  126: macdZeroCrossDown = ta.crossunder(macdLine, 0)
  127: 
  128: // --- ATR for Stop Loss and Take Profit ---
  129: atr = ta.atr(atrPeriod)
  130: stopLoss = atr * stopLossMultiplier
  131: takeProfit = atr * takeProfitMultiplier
  132: 
  133: // --- Trading Logic ---
  134: longCondition = macdZeroCrossUp and close > trendilo
  135: shortCondition = macdZeroCrossDown and close < trendilo
  136: 
  137: // --- Execute Long Trades ---
  138: if (longCondition)
  139:     strategy.entry("Buy", strategy.long)
  140:     strategy.exit("Take Profit/Stop Loss", "Buy", limit=close + takeProfit, stop=close - stopLoss)
  141: 
  142: // --- Execute Short Trades ---
  143: if (shortCondition)
  144:     strategy.entry("Sell", strategy.short)
  145:     strategy.exit("Take Profit/Stop Loss", "Sell", limit=close - takeProfit, stop=close + stopLoss)
