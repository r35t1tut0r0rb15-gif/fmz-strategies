## Target 1 % / stop 1 % of the signal close  [lines 201-222]
  201: percentTarget = input.float(1.0, title="Target %", minval=0.0, step=0.1) / 100
  202: percentStopLoss = input.float(1.0, title="Stop Loss %", minval=0.0, step=0.1) / 100
  203: 
  204: // Entry conditions
  205: longCondition = ta.crossover(close, supertrend)
  206: shortCondition = ta.crossunder(close, supertrend)
  207: 
  208: // Exit conditions
  209: takeProfitLevelLong = close * (1 + percentTarget)
  210: stopLossLevelLong = close * (1 - percentStopLoss)
  211: 
  212: takeProfitLevelShort = close * (1 - percentTarget)
  213: stopLossLevelShort = close * (1 + percentStopLoss)
  214: 
  215: // Execute trades
  216: if (longCondition)
  217:     strategy.entry("Long", strategy.long)
  218:     strategy.exit("Take Profit/Stop Loss", from_entry="Long", limit=takeProfitLevelLong, stop=stopLossLevelLong)
  219: 
  220: if (shortCondition)
  221:     strategy.entry("Short", strategy.short)
  222:     strategy.exit("Take Profit/Stop Loss", from_entry="Short", limit=takeProfitLevelShort, stop=stopLossLevelShort)
