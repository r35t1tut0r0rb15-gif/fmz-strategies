  205: //SL & TP Calculations
  206: SwingLow=lowest(i_SwingLookback)
  207: SwingHigh=highest(i_SwingLookback)
  208: bought=strategy.position_size != strategy.position_size[1]
  209: LSL=valuewhen(bought, SwingLow, 0)-((valuewhen(bought, atr(14), 0))*i_SLExpander)
  210: SSL=valuewhen(bought, SwingHigh, 0)+((valuewhen(bought, atr(14), 0))*i_SLExpander)
  211: lTP=strategy.position_avg_price + (strategy.position_avg_price-(valuewhen(bought, SwingLow, 0))+((valuewhen(bought, atr(14), 0))*i_TPExpander))
  212: sTP=strategy.position_avg_price - (valuewhen(bought, SwingHigh, 0)-strategy.position_avg_price)-((valuewhen(bought, atr(14), 0))*i_TPExpander)
  213: islong=strategy.position_size > 0
  214: isshort=strategy.position_size < 0
  215: SL= islong ? LSL : isshort ? SSL : na
  216: TP= islong ? lTP : isshort ? sTP : na
