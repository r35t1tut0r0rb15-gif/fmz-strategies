## strategy(): 250 % of equity; stop 1 % / target 2 % of the avg price  [lines 129-162]
  129: strategy("DMI + ADX Strategy", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=250)
  130: 
  131: // Nastavenie parametrov
  132: adxLength = input.int(14, title="ADX Length")
  133: adxSmoothing = input.int(14, title="ADX Smoothing")
  134: dmiLength = input.int(14, title="DMI Length")
  135: adxThreshold = input.float(25.0, title="ADX Threshold")
  136: stopLossPerc = input.float(1.0, title="Stop Loss (%)")
  137: takeProfitPerc = input.float(2.0, title="Take Profit (%)")
  138: 
  139: // Výpočet DMI a ADX pomocou ta.dmi
  140: [plusDI, minusDI, adxValue] = ta.dmi(dmiLength, adxSmoothing)
  141: 
  142: // Nákupné podmienky
  143: longCondition = ta.crossover(plusDI, minusDI) and adxValue > adxThreshold
  144: if (longCondition)
  145:     strategy.entry("Long", strategy.long)
  146: 
  147: // Predajné podmienky
  148: shortCondition = ta.crossunder(plusDI, minusDI) and adxValue > adxThreshold
  149: if (shortCondition)
  150:     strategy.entry("Short", strategy.short)
  151: 
  152: // Definovanie Stop a Limit pre Long pozíciu
  153: longStop = strategy.position_avg_price * (1 - stopLossPerc / 100)
  154: longLimit = strategy.position_avg_price * (1 + takeProfitPerc / 100)
  155: if (strategy.position_size > 0)
  156:     strategy.exit("Long Exit", "Long", stop=longStop, limit=longLimit)
  157: 
  158: // Definovanie Stop a Limit pre Short pozíciu
  159: shortStop = strategy.position_avg_price * (1 + stopLossPerc / 100)
  160: shortLimit = strategy.position_avg_price * (1 - takeProfitPerc / 100)
  161: if (strategy.position_size < 0)
  162:     strategy.exit("Short Exit", "Short", stop=shortStop, limit=shortLimit)
