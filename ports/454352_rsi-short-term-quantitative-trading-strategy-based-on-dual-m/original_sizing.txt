## Stop 1 ATR / target 2 ATR beyond the bar's low / high, every bar  [lines 118-141]
  118: risk = input(1, title="Riesgo en ATR")
  119: reward_ratio = input(2, title="Ratio Riesgo/Beneficio")
  120: atr_length = input(14, title="Periodo del ATR")
  121: atr = ta.atr(atr_length)
  122: stop_loss = risk * atr
  123: take_profit = reward_ratio * stop_loss

  125: // Señal de compra
