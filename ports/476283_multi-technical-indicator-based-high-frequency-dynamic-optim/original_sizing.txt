## strategy(): 10 % of equity; ATR 1.5 / 2.0 bracket from the avg price  [lines 92-151]
   92: strategy("Scalping BTC Ottimizzato - Grafica Chiara", shorttitle="Scalp BTC Opt", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=10)
   93: 
   94: // === ? INPUTS ===
   95: // ? Medie Mobili
   96: emaFastLength = input.int(9, title="EMA Veloce", minval=1)
   97: emaSlowLength = input.int(21, title="EMA Lenta", minval=1)
   98: 
   99: // ? RSI
  100: rsiLength = input.int(14, title="RSI Length", minval=1)
  101: rsiOverbought = input.int(70, title="RSI Overbought")
  102: rsiOversold = input.int(30, title="RSI Oversold")
  103: 
  104: // ? ATR (Stop Loss e Take Profit)
  105: atrLength = input.int(14, title="ATR Length", minval=1)
  106: stopATR = input.float(1.5, title="Stop Loss (ATR Multiplo)", step=0.1)
  107: takeProfitATR = input.float(2.0, title="Take Profit (ATR Multiplo)", step=0.1)
  108: 
  109: // ? ADX
  110: adxLength = input.int(14, title="ADX Length", minval=1)
  111: adxSmoothing = input.int(14, title="ADX Smoothing", minval=1)
  112: adxThreshold = input.int(20, title="Soglia ADX per Trend Forte", minval=1)
  113: 
  114: // === ? CALCOLI PRINCIPALI ===
  115: // ? Medie Mobili
  116: emaFast = ta.ema(close, emaFastLength)
  117: emaSlow = ta.ema(close, emaSlowLength)
  118: 
  119: // ? RSI
  120: rsi = ta.rsi(close, rsiLength)
  121: 
  122: // ? ATR
  123: atr = ta.atr(atrLength)
  124: 
  125: // ? ADX tramite DMI con Smoothing
  126: [adx, diPlus, diMinus] = ta.dmi(adxLength, adxSmoothing)
  127: 
  128: // === ? CONDIZIONI LONG E SHORT ===
  129: // ✅ Long: EMA Veloce incrocia EMA Lenta al rialzo, RSI sotto 70, ADX > 20
  130: longCondition = (ta.crossover(emaFast, emaSlow)) and (rsi < rsiOverbought) and (adx > adxThreshold)
  131: 
  132: // ? Short: EMA Veloce incrocia EMA Lenta al ribasso, RSI sopra 30, ADX > 20
  133: shortCondition = (ta.crossunder(emaFast, emaSlow)) and (rsi > rsiOversold) and (adx > adxThreshold)
  134: 
  135: // ? Stop Loss e Take Profit Dinamici
  136: longStop = strategy.position_avg_price - (atr * stopATR)
  137: longTarget = strategy.position_avg_price + (atr * takeProfitATR)
  138: 
  139: shortStop = strategy.position_avg_price + (atr * stopATR)
  140: shortTarget = strategy.position_avg_price - (atr * takeProfitATR)
  141: 
  142: // === ? INGRESSO E USCITA ===
  143: // ? Ingresso LONG
  144: if (longCondition and strategy.position_size == 0)
  145:     strategy.entry("Long", strategy.long)
  146:     strategy.exit("TakeProfit/StopLoss Long", stop=longStop, limit=longTarget)
  147: 
  148: // ? Ingresso SHORT
  149: if (shortCondition and strategy.position_size == 0)
  150:     strategy.entry("Short", strategy.short)
  151:     strategy.exit("TakeProfit/StopLoss Short", stop=shortStop, limit=shortTarget)
