## Bracket every bar  [lines 125-141]
  125: atr = ta.atr(atr_length)
  126: stop_loss = close - atr * atr_mult
  127: take_profit = close + atr * atr_mult
  128: // === MACD for Momentum Confirmation (Slow Settings for 4H) ===
  129: [macd_line, signal_line, _] = ta.macd(close, 26, 52, 18)
  130: macd_bullish = macd_line > signal_line
  131: macd_bearish = macd_line < signal_line
  132: // === Entry and Exit Conditions ===
  133: buy_condition = ta.crossover(close, kama) and macd_bullish
  134: sell_condition = ta.crossunder(close, kama) and macd_bearish
  135: // === Execute Trades ===
  136: if (buy_condition)
  137:     strategy.entry("Buy", strategy.long)
  138: if (sell_condition)
  139:     strategy.close("Buy")
  140: // === Dynamic Stop-Loss and Take-Profit ===
  141: strategy.exit("Exit", "Buy", stop=stop_loss, limit=take_profit)
