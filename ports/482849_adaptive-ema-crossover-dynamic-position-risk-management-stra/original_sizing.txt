   99: RiskPercent = input.float(1.0, title="Risk Percentage") // User-defined risk percentage
  100: 
  101: // Calculate EMAs

  103: Signal_LT_EMA = ta.ema(close, LT_EMA)

  105: // Plot EMAs

  110: uptrend = ta.crossover(Signal_MT_EMA, Signal_LT_EMA)
  111: downtrend = ta.crossunder(Signal_MT_EMA, Signal_LT_EMA)

  113: // Stop-Loss Calculations
  114: longStopLoss = ta.lowest(low, 2) // SL for buy = lowest low of last 2 candles
  115: shortStopLoss = ta.highest(high, 2) // SL for sell = highest high of last 2 candles
  116: 
  117: // Take-Profit Calculations
  118: longTakeProfit = close + (close - longStopLoss) * RR
  119: shortTakeProfit = close - (shortStopLoss - close) * RR

  121: // Calculate Position Size based on Risk Percentage
  122: capital = strategy.equity * (RiskPercent / 100)
  123: longPositionSize = capital / (close - longStopLoss)
  124: shortPositionSize = capital / (shortStopLoss - close)
