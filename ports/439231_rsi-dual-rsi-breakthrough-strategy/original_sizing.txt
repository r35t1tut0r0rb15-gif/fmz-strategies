  137: //Fast RSI
  138: fastup = rma(max(change(close), 0), fast)
  139: fastdown = rma(-min(change(close), 0), fast)
  140: fastrsi = fastdown == 0 ? 100 : fastup == 0 ? 0 : 100 - (100 / (1 + fastup / fastdown))
  141: 
  142: //Slow RSI
  143: slowup = rma(max(change(close), 0), slow)
  144: slowdown = rma(-min(change(close), 0), slow)
  145: slowrsi = slowdown == 0 ? 100 : slowup == 0 ? 0 : 100 - (100 / (1 + slowup / slowdown))
  146: 
  147: //Signals
  148: up = slowrsi > 50 and fastrsi < 50
  149: dn = slowrsi < 50 and fastrsi > 50
  150: exit = (strategy.position_size > 0 and close > open) or (strategy.position_size < 0 and close < open)
  151: lot = strategy.position_size == 0 ? strategy.equity / close * leverage : lot[1]
