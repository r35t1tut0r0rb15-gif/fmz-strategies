## Stop 1 %, target 2 R from the signal close  [lines 101-138]
  101: riskRewardRatio = input.float(2.0, title="Risk/Reward Ratio")
  102: stopLossPerc = input.float(1.0, title="Stop Loss Percentage") / 100
  103: 
  104: // Timeframe selection
  105: useTimeframe = input.timeframe("240", title="Timeframe for Analysis", options=["240", "D", "W", "M"])
  106: 
  107: // Request data from selected timeframe
  108: highTF = request.security(syminfo.tickerid, useTimeframe, high)
  109: lowTF = request.security(syminfo.tickerid, useTimeframe, low)
  110: 
  111: // Swing high and low calculation over the last 50 bars in the selected timeframe
  112: highestHigh = ta.highest(highTF, 50)
  113: lowestLow = ta.lowest(lowTF, 50)
  114: 
  115: // Fibonacci retracement levels
  116: fib618 = highestHigh - (highestHigh - lowestLow) * fiboLevel1
  117: fib786 = highestHigh - (highestHigh - lowestLow) * fiboLevel2
  118: 
  119: // Plot Fibonacci levels
  120: // line.new(bar_index[1], fib618, bar_index, fib618, color=color.red, width=2, style=line.style_dashed)
  121: // line.new(bar_index[1], fib786, bar_index, fib786, color=color.orange, width=2, style=line.style_dashed)
  122: 
  123: // Entry signals based on candlestick patterns and Fibonacci levels
  124: bullishCandle = close > open and close > fib618 and close < highestHigh
  125: bearishCandle = close < open and close < fib786 and close > lowestLow
  126: 
  127: // Stop loss and take profit calculation
  128: stopLoss = bullishCandle ? close * (1 - stopLossPerc) : close * (1 + stopLossPerc)
  129: takeProfit = bullishCandle ? close + (close - stopLoss) * riskRewardRatio : close - (stopLoss - close) * riskRewardRatio
  130: 
  131: // Plot buy and sell signals
  132: if bullishCandle
  133:     strategy.entry("Buy", strategy.long)
  134:     strategy.exit("Take Profit", "Buy", limit=takeProfit, stop=stopLoss)
  135: 
  136: if bearishCandle
  137:     strategy.entry("Sell", strategy.short)
  138:     strategy.exit("Take Profit", "Sell", limit=takeProfit, stop=stopLoss)
