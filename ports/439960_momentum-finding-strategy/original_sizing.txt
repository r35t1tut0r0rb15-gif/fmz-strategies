  180: // Calculate BB
  181: source = close
  182: basis = sma(source, length)
  183: dev = multKC * stdev(source, length)
  184: upperBB = basis + dev
  185: lowerBB = basis - dev
  186: 
  187: // Calculate KC
  188: ma = sma(source, lengthKC)
  189: range = useTrueRange ? tr : (high - low)
  190: rangema = sma(range, lengthKC)
  191: upperKC = ma + rangema * multKC
  192: lowerKC = ma - rangema * multKC
  193: 
  194: sqzOn  = (lowerBB > lowerKC) and (upperBB < upperKC)
  195: sqzOff = (lowerBB < lowerKC) and (upperBB > upperKC)
  196: noSqz  = (sqzOn == false) and (sqzOff == false)
