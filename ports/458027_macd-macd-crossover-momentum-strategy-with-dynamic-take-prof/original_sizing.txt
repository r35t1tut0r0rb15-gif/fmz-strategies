## Target 100 / stop 50 points from the average price  [lines 180-208]
  180: target_points = input.int(100, title="Target Points")
  181: stop_loss_points = input.int(50, title="Stop Loss Points")
  182: 
  183: // Calculate MACD

  185: 
  186: // Strategy logic
  187: long_condition = ta.crossover(macd_line, signal_line)
  188: short_condition = ta.crossunder(macd_line, signal_line)

  190: // Plot MACD
  191: plot(macd_line, color=color.blue, title="MACD Line")
  192: plot(signal_line, color=color.red, title="Signal Line")

  195: if long_condition
  196:     strategy.entry("Long", strategy.long)
  197: if short_condition
  198:     strategy.entry("Short", strategy.short)

  200: // Calculate target and stop loss levels
  201: long_target = strategy.position_avg_price + target_points
  202: long_stop_loss = strategy.position_avg_price - stop_loss_points
  203: short_target = strategy.position_avg_price - target_points
  204: short_stop_loss = strategy.position_avg_price + stop_loss_points

  206: // Strategy exit
  207: strategy.exit("Long Exit", "Long", limit=long_target, stop=long_stop_loss)
