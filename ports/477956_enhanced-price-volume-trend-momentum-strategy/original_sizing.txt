## qty 1; TP 10 x ATR, SL 3 x ATR  [lines 108-153]
  108: tpATRmult     = input.int(title="止盈 ATR 倍數 (1~10)", defval=10, minval=1, maxval=500)
  109: // 止損倍數 (1~10倍 ATR)
  110: slATRmult     = input.int(title="止損 ATR 倍數 (1~10)", defval=3, minval=1, maxval=500)
  111: 
  112: //=== 2) MACD 計算 ===//
  113: fast_ma  = sma_source == "SMA" ? ta.sma(src, fast_length) : ta.ema(src, fast_length)
  114: slow_ma  = sma_source == "SMA" ? ta.sma(src, slow_length) : ta.ema(src, slow_length)
  115: macd     = fast_ma - slow_ma
  116: signal   = sma_signal == "SMA" ? ta.sma(macd, signal_length) : ta.ema(macd, signal_length)
  117: hist     = macd - signal
  118: 
  119: //=== 3) 判斷深色/淺色（用於變化訊號）===//
  120: darkGreen  = hist >= 0 and hist <= hist[1]   // 上方，柱子縮小或持平
  121: lightGreen = hist >= 0 and hist >  hist[1]   // 上方，柱子變大
  122: darkRed    = hist <  0 and hist <= hist[1]   // 下方，柱子(絕對值)變大或持平
  123: lightRed   = hist <  0 and hist >  hist[1]   // 下方，柱子(絕對值)變小
  124: 
  125: // 由「深 → 淺」是否發生在上一根
  126: colorChangeToLightGreen = darkGreen[1] and lightGreen
  127: colorChangeToLightRed   = darkRed[1]   and lightRed
  128: 
  129: //=== 4) ATR 計算 (用於止盈止損) ===//
  130: atrPeriod  = 14
  131: atrValue   = ta.atr(atrPeriod)
  132: 
  133: //=== 5) 多單策略：深紅 → 淺紅 (底部紅色) ===//
  134: if useLong and colorChangeToLightRed
  135:     // 以當前 K 線 low - ATR倍數 作為多單止損
  136:     longStopLoss   = low - (slATRmult * atrValue)
  137:     // 以當前 close + ATR倍數 作為多單止盈
  138:     longTakeProfit = close + (tpATRmult * atrValue)
  139: 
  140:     // 進多單
  141:     strategy.entry("Long Entry", strategy.long, comment="多", qty=1)
  142:     strategy.exit("平多", "Long Entry", stop=longStopLoss, limit=longTakeProfit)
  143: 
  144: //=== 6) 空單策略：深綠 → 淺綠 (頂部綠色) ===//
  145: if useShort and colorChangeToLightGreen
  146:     // 以當前 K 線 high + ATR倍數 作為空單止損
  147:     shortStopLoss   = high + (slATRmult * atrValue)
  148:     // 以當前 close - ATR倍數 作為空單止盈
  149:     shortTakeProfit = close - (tpATRmult * atrValue)
  150: 
  151:     // 進空單
  152:     strategy.entry("Short Entry", strategy.short, comment="空", qty=1)
  153:     strategy.exit("平空", "Short Entry", stop=shortStopLoss, limit=shortTakeProfit)
