## Stop 1.5 % / target 3 % of the average price  [lines 108-126]
  108: stopLossPercent = 1.5  // Stop-loss at 1.5%
  109: takeProfitPercent = 3.0  // Take-profit at 3%
  110: 
  111: // Calculate stop-loss and take-profit levels for long and short positions
  112: longStopLoss = strategy.position_avg_price * (1 - stopLossPercent / 100)
  113: longTakeProfit = strategy.position_avg_price * (1 + takeProfitPercent / 100)
  114: 
  115: shortStopLoss = strategy.position_avg_price * (1 + stopLossPercent / 100)
  116: shortTakeProfit = strategy.position_avg_price * (1 - takeProfitPercent / 100)
  117: 
  118: // Enter long position with stop-loss and take-profit
  119: if (longCondition)
  120:     strategy.entry("Long", strategy.long)
  121:     strategy.exit("Take Profit/Stop Loss", "Long", stop=longStopLoss, limit=longTakeProfit)
  122: 
  123: // Enter short position with stop-loss and take-profit
  124: if (shortCondition)
  125:     strategy.entry("Short", strategy.short)
  126:     strategy.exit("Take Profit/Stop Loss", "Short", stop=shortStopLoss, limit=shortTakeProfit)
