## strategy(): capital 100000, 250 % of equity; trailing stop at the previous low / high  [lines 91-173]
   91: strategy("TrinityBar with Trailing Stop", overlay=true, initial_capital=100000, 
   92:      default_qty_type=strategy.percent_of_equity, default_qty_value=250)

   94: //─────────────────────────────────────────────────────────────
   95: // 1. BAR THIRDS CALCULATIONS

   97: cur_range      = high - low
   98: cur_lowerThird = low + cur_range / 3
   99: cur_upperThird = high - cur_range / 3
  100: 
  101: prev_range      = high[1] - low[1]
  102: prev_lowerThird = low[1] + prev_range / 3
  103: prev_upperThird = high[1] - prev_range / 3

  105: //─────────────────────────────────────────────────────────────
  106: // 2. DEFINE BULLISH & BEARISH BAR TYPES (CURRENT & PREVIOUS)
  107: //─────────────────────────────────────────────────────────────
  108: // Current bar types
  109: is_1_3 = (open <= cur_lowerThird) and (close >= cur_upperThird)
  110: is_3_3 = (open >= cur_upperThird) and (close >= cur_upperThird)
  111: is_2_3 = (open > cur_lowerThird) and (open < cur_upperThird) and (close >= cur_upperThird)
  112: 
  113: is_3_1 = (open >= cur_upperThird) and (close <= cur_lowerThird)
  114: is_1_1 = (open <= cur_lowerThird) and (close <= cur_lowerThird)
  115: is_2_1 = (open > cur_lowerThird) and (open < cur_upperThird) and (close <= cur_lowerThird)
  116: 
  117: // Previous bar types
  118: prev_is_1_3 = (open[1] <= prev_lowerThird) and (close[1] >= prev_upperThird)
  119: prev_is_3_3 = (open[1] >= prev_upperThird) and (close[1] >= prev_upperThird)
  120: prev_is_2_3 = (open[1] > prev_lowerThird) and (open[1] < prev_upperThird) and (close[1] >= prev_upperThird)
  121: 
  122: prev_is_3_1 = (open[1] >= prev_upperThird) and (close[1] <= prev_lowerThird)
  123: prev_is_1_1 = (open[1] <= prev_lowerThird) and (close[1] <= prev_lowerThird)
  124: prev_is_2_1 = (open[1] > prev_lowerThird) and (open[1] < prev_upperThird) and (close[1] <= prev_lowerThird)
  125: 
  126: //─────────────────────────────────────────────────────────────
  127: // 3. VALID SIGNAL CONDITIONS
  128: //─────────────────────────────────────────────────────────────
  129: validBuy  = (prev_is_2_3 or prev_is_3_3 or prev_is_1_3) and (is_1_3 or is_3_3)
  130: validSell = (prev_is_2_1 or prev_is_1_1 or prev_is_3_1) and (is_1_1 or is_3_1)
  131: 
  132: //─────────────────────────────────────────────────────────────
  133: // 4. PLOT SIGNAL TRIANGLES
  134: //─────────────────────────────────────────────────────────────
  135: plotshape(validBuy, title="Valid Buy", style=shape.triangleup, location=location.belowbar,

  137: plotshape(validSell, title="Valid Sell", style=shape.triangledown, location=location.abovebar, 
  138:      color=color.red, size=size.small, text="S")

  140: //─────────────────────────────────────────────────────────────
  141: // 5. MARKET ORDER EXECUTION BASED ON SIGNALS
  142: //─────────────────────────────────────────────────────────────
  143: if validBuy
  144:     // Close any short positions.
  145:     strategy.close("Short", comment="")
  146:     // If not already long, enter a market long.
  147:     if strategy.position_size <= 0
  148:         strategy.entry("Long", strategy.long, comment="")
  149:         
  150: if validSell
  151:     // Close any long positions.
  152:     strategy.close("Long", comment="")
  153:     // If not already short, enter a market short.
  154:     if strategy.position_size >= 0
  155:         strategy.entry("Short", strategy.short, comment="")

  158: // 6. TRAILING STOP LOSS FUNCTION
  159: //─────────────────────────────────────────────────────────────
  160: // Inputs for trailing stop settings:
  161: trailBars = input.int(title="Trailing Stop Bars Back", defval=1, minval=1)
  162: trailTF   = input.timeframe(title="Trailing Stop Timeframe", defval="")  // "" = current timeframe

  164: // For long positions, use the low from 'trailBars' bars back on the specified timeframe.
  165: // For short positions, use the high from 'trailBars' bars back.
  166: trailStopLong  = request.security(syminfo.tickerid, trailTF, low[trailBars])
  167: trailStopShort = request.security(syminfo.tickerid, trailTF, high[trailBars])
  168: 
  169: // Apply trailing stops if a position is open.
  170: if strategy.position_size > 0
  171:     strategy.exit("Trailing Stop Long", from_entry="Long", stop=trailStopLong)
  172: if strategy.position_size < 0
  173:     strategy.exit("Trailing Stop Short", from_entry="Short", stop=trailStopShort)
