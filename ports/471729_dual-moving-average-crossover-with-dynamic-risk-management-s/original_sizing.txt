## Stop 2.5 % / target 4 % of the signal close (close tests)  [lines 103-125]
  103: stop_loss_percent = input.float(2.5, title="Stop-Loss (%)", step=0.1) // Stop-Loss moderado
  104: take_profit_percent = input.float(4.0, title="Take-Profit (%)", step=0.1) // Take-Profit moderado
  105: 
  106: // Cálculo de medias móviles (Exponenciales)
  107: mmr = ta.ema(close, mmr_period) // Media Móvil Rápida
  108: mml = ta.ema(close, mml_period) // Media Móvil Lenta
  109: 
  110: // Señales de Compra y Venta
  111: long_condition = ta.crossover(mmr, mml)  // Señal de compra
  112: short_condition = ta.crossunder(mmr, mml) // Señal de venta
  113: 
  114: // Calcular niveles de Stop-Loss y Take-Profit solo al activar la compra
  115: var float entry_price = na
  116: var float stop_loss_level = na
  117: var float take_profit_level = na
  118: 
  119: if (long_condition)
  120:     entry_price := close
  121:     stop_loss_level := entry_price * (1 - stop_loss_percent / 100)
  122:     take_profit_level := entry_price * (1 + take_profit_percent / 100)
  123: 
  124: // Condiciones de salida (Stop-Loss y Take-Profit)
  125: exit_condition = (close <= stop_loss_level) or (close >= take_profit_level)
