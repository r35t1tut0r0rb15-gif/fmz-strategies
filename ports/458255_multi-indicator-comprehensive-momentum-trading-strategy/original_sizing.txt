## Target / stop 1.5 ATR from the signal close  [lines 180-208]
  180: atrLength = input.int(14, title="ATR Length")
  181: atrMultiplier = input.float(1.5, title="ATR Multiplier")
  182: 
  183: // EMA Calculation
  184: emaShort = ta.ema(close, emaShortLength)
  185: emaLong = ta.ema(close, emaLongLength)
  186: 
  187: // MACD Calculation
  188: [macdLine, signalLine, _] = ta.macd(close, macdFastLength, macdSlowLength, macdSignalSmoothing)
  189: macdHist = macdLine - signalLine
  190: 
  191: // RSI Calculation
  192: rsi = ta.rsi(close, rsiLength)
  193: 
  194: // ATR Calculation
  195: atr = ta.atr(atrLength)
  196: 
  197: // Trading Conditions
  198: longCondition = emaShort > emaLong and macdLine > signalLine and rsi < rsiOverbought
  199: shortCondition = emaShort < emaLong and macdLine < signalLine and rsi > rsiOversold
  200: 
  201: // Trade Execution with Risk Management
  202: if (longCondition)
  203:     strategy.entry("Long", strategy.long)
  204:     strategy.exit("Take Profit/Stop Loss", from_entry="Long", limit=close + atr * atrMultiplier, stop=close - atr * atrMultiplier)
  205: 
  206: if (shortCondition)
  207:     strategy.entry("Short", strategy.short)
  208:     strategy.exit("Take Profit/Stop Loss", from_entry="Short", limit=close - atr * atrMultiplier, stop=close + atr * atrMultiplier)
