## Limit 0.98 % from the current close (re-sent every bar)  [lines 131-159]
  131: porcentaje = input.float(0.98, title="Porcentaje de Salida de Banda")
  132: 
  133: // Definir señales de compra y venta
  134: compra = ta.crossover(close, banda_inferior)
  135: venta = ta.crossunder(close, banda_superior)
  136: 
  137: // Calcular el precio de salida para las señales de compra y venta
  138: precio_salida_compra = close * (1 + porcentaje / 100)
  139: precio_salida_venta = close * (1 - porcentaje / 100)
  140: 
  141: // Plotear las bandas
  142: plot(banda_superior, color=color.blue, linewidth=2, title="Banda Superior")
  143: plot(banda_inferior, color=color.red, linewidth=2, title="Banda Inferior")
  144: 
  145: // Plotear las señales de compra y venta
  146: plotshape(compra, style=shape.triangleup, location=location.belowbar, color=color.green, size=size.small, title="Compra")
  147: plotshape(venta, style=shape.triangledown, location=location.abovebar, color=color.red, size=size.small, title="Venta")
  148: 
  149: // Simular operaciones
  150: if (compra)
  151:     strategy.entry("Compra", strategy.long)
  152: if (venta)
  153:     strategy.entry("Venta", strategy.short)
  154: 
  155: // Definir reglas de salida
  156: if (strategy.position_size > 0)
  157:     strategy.exit("Exit Long", from_entry="Compra", limit=precio_salida_compra)
  158: if (strategy.position_size < 0)
  159:     strategy.exit("Exit Short", from_entry="Venta", limit=precio_salida_venta)
