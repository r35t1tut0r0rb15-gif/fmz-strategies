## Leverage input and the leveraged contract quantity  [lines 215-221]
  215: leverage=input(2,"leverage",step=.5)
  216: enableshort=input(true,"enable shorts?")
  217: 
  218: barcount=0
  219: barcount:=nz(barcount[1])+1
  220: 
  221: contracts=min(max(.000001,(strategy.equity/close)*leverage),50000)

## Orders with the quantity  [lines 222-228]
  222: strategy.entry("Long",strategy.long,when=long and barcount>20, qty=contracts)
  223: 
  224: strategy.close("Long",when=short or end )
  225: 
  226: strategy.entry("Short",strategy.short,when=short and enableshort==true and barcount>20, qty=contracts)
  227: 
  228: strategy.close("Short",when=(long or end) and enableshort==true)
