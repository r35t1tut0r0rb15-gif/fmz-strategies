  111: //#region STRATEGY 2 INPUTS
  112: s2_fastMALen = input.int(defval=20, title="Fast SMA Length (S2)", minval=1, group="Strategy 2 Settings", inline="S2 MA")
  113: s2_slowMALen = input.int(defval=50, title="Slow SMA Length (S2)", minval=1, group="Strategy 2 Settings", inline="S2 MA")

  122: var float stopLossPrice = na
  123: var float trailingStopPrice = na

  125: var float slowMA = na

  127: var bool validOpenLongPosition = false
