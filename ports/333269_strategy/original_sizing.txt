## Cancel-all and cover helpers  [lines 53-82]
   53: function cancellAll() {
   54:     while (true) {
   55:         var orders = _C(exchange.GetOrders)
   56:         if (orders.length == 0) {
   57:             break
   58:         } else {
   59:             for (var i = 0 ; i < orders.length ; i++) {
   60:                 exchange.CancelOrder(orders[i].Id, orders[i])
   61:                 Sleep(500)
   62:             }
   63:         }
   64:         Sleep(500)
   65:     }
   66: }

   68: function cover(tradeFunc, direction) {
   69:     var mapDirection = {"closebuy": PD_LONG, "closesell": PD_SHORT}
   70:     var positions = _C(exchange.GetPosition)
   71:     var pos = getPosition(positions, mapDirection[direction])
   72:     if (pos.Amount > 0) {
   73:         cancellAll()
   74:         exchange.SetDirection(direction)
   75:         if (tradeFunc(-1, pos.Amount)) {
   76:             return true 
   77:         } else {
   78:             return false 
   79:         }
   80:     }
   81:     return true 
   82: }
