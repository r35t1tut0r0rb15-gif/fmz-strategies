## TP 0.5 x / SL 1.0 x the outside bar height  [lines 133-152]

  135: // Calculate TP and SL levels

  142: tpLevelShort = close - outsideBarHeight * tpRatio  // TP cho lệnh bán
  143: slLevelShort = close + outsideBarHeight * slRatio  // SL cho lệnh bán
