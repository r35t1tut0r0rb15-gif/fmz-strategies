  273: tema(src, len) =>
  274:     ema1 = ta.ema(src, len)
  275:     ema2 = ta.ema(ema1, len)
  276:     ema3 = ta.ema(ema2, len)
  277:     3 * ema1 - 3 * ema2 + ema3
  278: ////// EDSMA
  279: get2PoleSSF(src, length) =>
  280:     PI = 2 * math.asin(1)
  281:     arg = math.sqrt(2) * PI / length
  282:     a1 = math.exp(-arg)
  283:     b1 = 2 * a1 * math.cos(arg)
  284:     c2 = b1
  285:     c3 = -math.pow(a1, 2)
  286:     c1 = 1 - c2 - c3
  287: 
  288:     ssf = 0.0
  289:     ssf := c1 * src + c2 * nz(ssf[1]) + c3 * nz(ssf[2])
  290:     ssf
  291: 
  292: get3PoleSSF(src, length) =>
  293:     PI = 2 * math.asin(1)
  294: 
  295:     arg = PI / length
  296:     a1 = math.exp(-arg)
  297:     b1 = 2 * a1 * math.cos(1.738 * arg)
  298:     c1 = math.pow(a1, 2)
  299: 
  300:     coef2 = b1 + c1
  301:     coef3 = -(c1 + b1 * c1)
  302:     coef4 = math.pow(c1, 2)
  303:     coef1 = 1 - coef2 - coef3 - coef4
  304: 
  305:     ssf = 0.0
  306:     ssf := coef1 * src + coef2 * nz(ssf[1]) + coef3 * nz(ssf[2]) + coef4 * nz(ssf[3])
  307:     ssf
  308: 
  309: ma(type, src, len) =>
  310:     float result = 0
  311:     if type == 'TMA'
  312:         result := ta.sma(ta.sma(src, math.ceil(len / 2)), math.floor(len / 2) + 1)
  313:         result
  314:     if type == 'MF'
  315:         ts = 0.
  316:         b = 0.
  317:         c = 0.
  318:         os = 0.
  319:         //----
  320:         alpha = 2 / (len + 1)
  321:         a = i_feedback ? i_z * src + (1 - i_z) * nz(ts[1], src) : src
  322:         //----
  323:         b := a > alpha * a + (1 - alpha) * nz(b[1], a) ? a : alpha * a + (1 - alpha) * nz(b[1], a)
  324:         c := a < alpha * a + (1 - alpha) * nz(c[1], a) ? a : alpha * a + (1 - alpha) * nz(c[1], a)
  325:         os := a == b ? 1 : a == c ? 0 : os[1]
  326:         //----
  327:         upper = i_beta * b + (1 - i_beta) * c
  328:         lower = i_beta * c + (1 - i_beta) * b
  329:         ts := os * upper + (1 - os) * lower
  330:         result := ts
  331:         result
  332:     if type == 'LSMA'
  333:         result := ta.linreg(src, len, 0)
  334:         result
  335:     if type == 'SMA'  // Simple
  336:         result := ta.sma(src, len)
  337:         result
  338:     if type == 'EMA'  // Exponential
  339:         result := ta.ema(src, len)
  340:         result
  341:     if type == 'DEMA'  // Double Exponential
  342:         e = ta.ema(src, len)
  343:         result := 2 * e - ta.ema(e, len)
  344:         result
  345:     if type == 'TEMA'  // Triple Exponential
  346:         e = ta.ema(src, len)
  347:         result := 3 * (e - ta.ema(e, len)) + ta.ema(ta.ema(e, len), len)
  348:         result
  349:     if type == 'WMA'  // Weighted
  350:         result := ta.wma(src, len)
  351:         result
  352:     if type == 'VAMA'  // Volatility Adjusted
  353:         /// Copyright © 2019 to present, Joris Duyck (JD)
  354:         mid = ta.ema(src, len)
  355:         dev = src - mid
  356:         vol_up = ta.highest(dev, i_volatility_lookback)
  357:         vol_down = ta.lowest(dev, i_volatility_lookback)
  358:         result := mid + math.avg(vol_up, vol_down)
  359:         result
  360:     if type == 'HMA'  // Hull
  361:         result := ta.wma(2 * ta.wma(src, len / 2) - ta.wma(src, len), math.round(math.sqrt(len)))
  362:         result
  363:     if type == 'JMA'  // Jurik
  364:         /// Copyright © 2018 Alex Orekhov (everget)
  365:         /// Copyright © 2017 Jurik Research and Consulting.
  366:         phaseRatio = i_jurik_phase < -100 ? 0.5 : i_jurik_phase > 100 ? 2.5 : i_jurik_phase / 100 + 1.5
  367:         beta = 0.45 * (len - 1) / (0.45 * (len - 1) + 2)
  368:         alpha = math.pow(beta, i_jurik_power)
  369:         jma = 0.0
  370:         e0 = 0.0
  371:         e0 := (1 - alpha) * src + alpha * nz(e0[1])
  372:         e1 = 0.0
  373:         e1 := (src - e0) * (1 - beta) + beta * nz(e1[1])
  374:         e2 = 0.0
  375:         e2 := (e0 + phaseRatio * e1 - nz(jma[1])) * math.pow(1 - alpha, 2) + math.pow(alpha, 2) * nz(e2[1])
  376:         jma := e2 + nz(jma[1])
  377:         result := jma
  378:         result
  379:     if type == 'Kijun v2'
  380:         kijun = math.avg(ta.lowest(len), ta.highest(len))  //, (open + close)/2)
  381:         conversionLine = math.avg(ta.lowest(len / i_kidiv), ta.highest(len / i_kidiv))
  382:         delta = (kijun + conversionLine) / 2
  383:         result := delta
  384:         result
  385:     if type == 'McGinley'
  386:         mg = 0.0
  387:         mg := na(mg[1]) ? ta.ema(src, len) : mg[1] + (src - mg[1]) / (len * math.pow(src / mg[1], 4))
  388:         result := mg
  389:         result
  390:     if type == 'EDSMA'
  391:         zeros = src - nz(src[2])
  392:         avgZeros = (zeros + zeros[1]) / 2
  393: 
  394:         // Ehlers Super Smoother Filter 
  395:         ssf = i_ssfPoles == 2 ? get2PoleSSF(avgZeros, i_ssfLength) : get3PoleSSF(avgZeros, i_ssfLength)
  396: 
  397:         // Rescale filter in terms of Standard Deviations
  398:         stdev = ta.stdev(ssf, len)
  399:         scaledFilter = stdev != 0 ? ssf / stdev : 0
  400: 
  401:         alpha = 5 * math.abs(scaledFilter) / len
  402: 
  403:         edsma = 0.0
  404:         edsma := alpha * src + (1 - alpha) * nz(edsma[1])
  405:         result := edsma
  406:         result
  407:     result
  408:
