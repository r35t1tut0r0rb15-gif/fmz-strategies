## qty 1; take profit 0.02 / stop 0.01 (fractions)  [lines 96-128]
   96: take_profit_percent = input.float(0.02, title="Take Profit (%)")
   97: stop_loss_percent = input.float(0.01, title="Stop Loss (%)")
   98: risk_adjustment_factor = input.float(1.5, title="Risk Adjustment Factor")
   99: 
  100: // Fetch Exogenous Variable (Example: RSI as a Proxy)
  101: rsi_value = ta.rsi(close, lookback_period)
  102: 
  103: // Linear Signal Calculation
  104: linear_signal = signal_alpha * rsi_value + (1 - signal_alpha) * close
  105: 
  106: // Z-Score Normalization for Signal
  107: mean_signal = ta.sma(linear_signal, lookback_period)
  108: stddev_signal = ta.stdev(linear_signal, lookback_period)
  109: z_score_signal = (linear_signal - mean_signal) / stddev_signal
  110: 
  111: // Entry Conditions
  112: long_condition = z_score_signal < -risk_adjustment_factor
  113: short_condition = z_score_signal > risk_adjustment_factor
  114: 
  115: // Risk Management
  116: long_take_profit = close * (1 + take_profit_percent)
  117: long_stop_loss = close * (1 - stop_loss_percent)
  118: short_take_profit = close * (1 - take_profit_percent)
  119: short_stop_loss = close * (1 + stop_loss_percent)
  120: 
  121: // Execute Trades
  122: if (long_condition)
  123:     strategy.entry("Long", strategy.long, qty=1)
  124:     strategy.exit("Exit Long", "Long", stop=long_stop_loss, limit=long_take_profit)
  125: 
  126: if (short_condition)
  127:     strategy.entry("Short", strategy.short, qty=1)
  128:     strategy.exit("Exit Short", "Short", stop=short_stop_loss, limit=short_take_profit)
