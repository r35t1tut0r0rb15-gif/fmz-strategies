## Take-profit 1.5 % (computed, never sent)  [lines 106-131]

  108: // Индикаторы EMA (ускоренные для более частых сигналов)

  112: // MACD с более короткими периодами для большей чувствительности
  113: [macd_line, signal_line, _] = ta.macd(close, macd_short, macd_long, macd_signal)
  114: 
  115: // Индикатор RSI с упрощенными уровнями для большего количества сигналов
  116: rsi = ta.rsi(close, rsi_length)

  118: // Сигналы на покупку и продажу с ослабленными условиями
  119: buy_signal = ta.crossover(macd_line, signal_line) and ema_20 > ema_50 and rsi > 40
  120: sell_signal = ta.crossunder(macd_line, signal_line) and ema_20 < ema_50 and rsi < 60
  121: 
  122: // Логика открытия сделок и расчет тейк-профита
  123: var float take_profit_price = na  // переменная для хранения уровня тейк-профита
  124: 
  125: if (buy_signal)
  126:     strategy.entry("Buy", strategy.long)
  127:     take_profit_price := close * (1 + take_profit_percent / 100)  // уровень тейк-профита для покупки

  129: if (sell_signal)
  130:     strategy.entry("Sell", strategy.short)
  131:     take_profit_price := close * (1 - take_profit_percent / 100)  // уровень тейк-профита для продажи
