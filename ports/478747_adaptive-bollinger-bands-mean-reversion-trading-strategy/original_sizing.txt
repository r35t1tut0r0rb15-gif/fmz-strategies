## strategy(): 200 % of equity; stop 1 % / target 2 %  [lines 119-161]
  119: strategy("Bollinger Bands Strategy", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=200)
  120: 
  121: // Inputs for Bollinger Bands
  122: bbLength = input.int(20, title="Bollinger Bands Length")
  123: bbStdDev = input.float(2.0, title="Bollinger Bands StdDev")
  124: 
  125: // Inputs for Risk Management
  126: stopLossPerc = input.float(1.0, title="Stop Loss (%)", minval=0.1, step=0.1)
  127: takeProfitPerc = input.float(2.0, title="Take Profit (%)", minval=0.1, step=0.1)
  128: 
  129: // Calculate Bollinger Bands
  130: basis = ta.sma(close, bbLength)
  131: bbStdev = ta.stdev(close, bbLength)
  132: upper = basis + bbStdDev * bbStdev
  133: lower = basis - bbStdDev * bbStdev
  134: 
  135: // Plot Bollinger Bands
  136: plot(basis, color=color.blue, title="Middle Band")
  137: plot(upper, color=color.red, title="Upper Band")
  138: plot(lower, color=color.green, title="Lower Band")
  139: 
  140: // Entry Conditions
  141: longCondition = ta.crossover(close, lower)
  142: shortCondition = ta.crossunder(close, upper)
  143: 
  144: // Exit Conditions
  145: exitLongCondition = ta.crossunder(close, basis)
  146: exitShortCondition = ta.crossover(close, basis)
  147: 
  148: // Stop Loss and Take Profit Levels
  149: longStopLoss = close * (1 - stopLossPerc / 100)
  150: longTakeProfit = close * (1 + takeProfitPerc / 100)
  151: shortStopLoss = close * (1 + stopLossPerc / 100)
  152: shortTakeProfit = close * (1 - takeProfitPerc / 100)
  153: 
  154: // Execute Long Trades
  155: if (longCondition)
  156:     strategy.entry("Long", strategy.long)
  157:     strategy.exit("Exit Long", from_entry="Long", stop=longStopLoss, limit=longTakeProfit)
  158: 
  159: if (shortCondition)
  160:     strategy.entry("Short", strategy.short)
  161:     strategy.exit("Exit Short", from_entry="Short", stop=shortStopLoss, limit=shortTakeProfit)
