## Stop EMA 50 -+ 100 pips, target close +- 200 pips (x mintick)  [lines 178-223]
  178: tp_pips = input.int(200, title="Take Profit em Pips")
  179: sl_pips = input.int(100, title="Stop Loss em Pips")
  180: 
  181: // Cálculo das médias móveis
  182: ema_short = ta.ema(close, ema_short_length)
  183: ema_long = ta.ema(close, ema_long_length)
  184: 
  185: // Definição do Take Profit e Stop Loss iniciais em pips
  186: pip_size = syminfo.mintick
  187: initial_take_profit_buy = tp_pips * pip_size
  188: initial_take_profit_sell = tp_pips * pip_size
  189: initial_stop_loss_buy = ema_long - sl_pips * pip_size
  190: initial_stop_loss_sell = ema_long + sl_pips * pip_size
  191: 
  192: // Variáveis para controle de SL e TP móveis
  193: var float stop_loss_level = na
  194: var float take_profit_level = na
  195: 
  196: // Condições para Compra e Venda
  197: buy_condition = ta.crossover(ema_short, ema_long)
  198: sell_condition = ta.crossunder(ema_short, ema_long)
  199: 
  200: // Atualização do Stop Loss Móvel e Take Profit Móvel
  201: if (buy_condition)
  202:     stop_loss_level := ema_long - sl_pips * pip_size
  203:     take_profit_level := close + initial_take_profit_buy
  204: 
  205: if (sell_condition)
  206:     stop_loss_level := ema_long + sl_pips * pip_size
  207:     take_profit_level := close - initial_take_profit_sell
  208: 
  209: // Execução da Estratégia de Compra
  210: if (buy_condition)
  211:     strategy.entry("Compra", strategy.long)
  212: 
  213: // Saída da Estratégia de Compra
  214: if (strategy.position_size > 0)
  215:     strategy.exit("Take Profit", "Compra", limit=take_profit_level, stop=stop_loss_level)
  216: 
  217: // Execução da Estratégia de Venda
  218: if (sell_condition)
  219:     strategy.entry("Venda", strategy.short)
  220: 
  221: // Saída da Estratégia de Venda
  222: if (strategy.position_size < 0)
  223:     strategy.exit("Take Profit", "Venda", limit=take_profit_level, stop=stop_loss_level)
