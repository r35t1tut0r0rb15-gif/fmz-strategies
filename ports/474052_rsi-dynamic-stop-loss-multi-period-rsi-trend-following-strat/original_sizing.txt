## Size from a typed balance, risk 1.5 %, leverage 2; target 4 ATR / stop 2 ATR  [lines 94-135]
   94: account_balance = input.float(28.37, title="Account Balance", tooltip="Update this with your balance")
   95: risk_per_trade = input.float(0.015, title="Risk per Trade", tooltip="1.5% risk")
   96: leverage = input.int(2, title="Leverage", minval=1)
   97: stop_loss_percentage = input.float(0.015, title="Stop Loss Percentage", tooltip="1.5% stop loss")
   98: take_profit_multiplier = input.float(4, title="Take Profit Multiplier", tooltip="Take Profit is 4x Stop Loss")
   99: stop_loss_multiplier = input.float(2, title="Stop Loss Multiplier", tooltip="Dynamic Stop Loss Multiplier")
  100: 
  101: // Trade Size Calculation
  102: position_size = account_balance * risk_per_trade / (stop_loss_percentage / leverage)
  103: trade_qty = position_size / close // This gives you the qty in terms of contracts
  104: 
  105: // Indicators
  106: rsiLength = input.int(14, title="RSI Length")
  107: emaShort = input.int(9, title="Short-term EMA Length")
  108: emaLong = input.int(21, title="Long-term EMA Length")
  109: rsi = ta.rsi(close, rsiLength)
  110: emaShortLine = ta.ema(close, emaShort)
  111: emaLongLine = ta.ema(close, emaLong)
  112: 
  113: // Entry Conditions
  114: longCondition = ta.crossover(emaShortLine, emaLongLine) and rsi < 70
  115: shortCondition = ta.crossunder(emaShortLine, emaLongLine) and rsi > 30
  116: 
  117: // ATR for dynamic stop loss and take profit levels
  118: atrLength = input.int(14, title="ATR Length")
  119: atrMultiplier = input.float(1.5, title="ATR Multiplier")
  120: atr = ta.atr(atrLength)
  121: 
  122: // Dynamic Take Profit and Stop Loss Levels
  123: longTakeProfitLevel = close + (atr * take_profit_multiplier)
  124: longStopLossLevel = close - (atr * stop_loss_multiplier)
  125: shortTakeProfitLevel = close - (atr * take_profit_multiplier)
  126: shortStopLossLevel = close + (atr * stop_loss_multiplier)
  127: 
  128: // Strategy Execution
  129: if (longCondition)
  130:     strategy.entry("Long", strategy.long, qty=trade_qty)
  131:     strategy.exit("Take Profit/Stop Loss", from_entry="Long", limit=longTakeProfitLevel, stop=longStopLossLevel)
  132: 
  133: if (shortCondition)
  134:     strategy.entry("Short", strategy.short, qty=trade_qty)
  135:     strategy.exit("Take Profit/Stop Loss", from_entry="Short", limit=shortTakeProfitLevel, stop=shortStopLossLevel)
