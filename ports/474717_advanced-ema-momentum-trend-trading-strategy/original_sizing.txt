  152:     start_time = timestamp(year, month, dayofmonth, start_hour, start_minute, 0)
  153:     end_time = timestamp(year, month, dayofmonth, end_hour, end_minute, 0)
  154:     in_session = current_time >= start_time and current_time <= end_time
  155:     not use_time_filter or in_session

  171: emaUpTrend = close > ema200
  172: emaDownTrend = close < ema200

  174: // Trading Conditions

  176: shortCondition = crossoverDown and (not useEmaFilter or emaDownTrend) and is_within_session() and dailyTradeCount < maxDailyTrades and barstate.isconfirmed

  178: // Calcul des niveaux de Stop Loss et Take Profit
  179: float stopLoss = useAtrSl ? (atr * atrMultiplier) : (close * stopLossPerc / 100)
  180: float takeProfit = stopLoss * tpRatio
  181: 
  182: // Modification des variables pour éviter les erreurs de repainting
  183: var float entryPrice = na
  184: var float currentStopLoss = na
  185: var float currentTakeProfit = na
  186: 
  187: // Exécution des ordres avec gestion des positions

  191:         currentStopLoss := entryPrice - stopLoss
  192:         currentTakeProfit := entryPrice + takeProfit

  194:         strategy.exit("Exit Long", "Long", stop=currentStopLoss, limit=currentTakeProfit)
  195:         dailyTradeCount += 1
  196: 
  197:     if shortCondition
  198:         entryPrice := close
  199:         currentStopLoss := entryPrice + stopLoss
  200:         currentTakeProfit := entryPrice - takeProfit
