  216: // Smooth Average Range
  217: smoothrng(x, t, m) =>
  218:     wper = t * 2 - 1
  219:     avrng = ta.ema(math.abs(x - x[1]), t)
  220:     smoothrng = ta.ema(avrng, wper) * m
  221:     smoothrng
  222: smrng = smoothrng(src, per, mult)
  223: 
  224: // Range Filter
  225: rngfilt(x, r) =>
  226:     rngfilt = x
  227:     rngfilt := x > nz(rngfilt[1]) ? x - r < nz(rngfilt[1]) ? nz(rngfilt[1]) : x - r : 
  228:        x + r > nz(rngfilt[1]) ? nz(rngfilt[1]) : x + r
  229:     rngfilt
  230: filt = rngfilt(src, smrng)
  231: 
  232: // Filter Direction
  233: upward = 0.0
  234: upward := filt > filt[1] ? nz(upward[1]) + 1 : filt < filt[1] ? 0 : nz(upward[1])
  235: downward = 0.0
  236: downward := filt < filt[1] ? nz(downward[1]) + 1 : filt > filt[1] ? 0 : nz(downward[1])

  238: // Target Bands

  290: DEAD_ZONE = nz(ta.rma(ta.tr(true), 100)) * 3.7
  291: 
  292: calc_macd(source, fastLength, slowLength) =>
  293:     fastMA = ta.ema(source, fastLength)
  294:     slowMA = ta.ema(source, slowLength)
  295:     fastMA - slowMA
  296: 
  297: calc_BBUpper(source, length, multt) =>
  298:     basis = ta.sma(source, length)
  299:     dev = multt * ta.stdev(source, length)
  300:     basis + dev
  301: 
  302: calc_BBLower(source, length, multt) =>
  303:     basis = ta.sma(source, length)
  304:     dev = multt * ta.stdev(source, length)
  305:     basis - dev
