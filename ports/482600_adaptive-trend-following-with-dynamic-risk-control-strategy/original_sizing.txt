## ATR / target inputs  [lines 110-117]
  110: atrLength = input.int(14, "ATR 計算週期")
  111: atrMultiplier = input.float(1.3, "ATR 係數")  // 2倍 ATR 作為止損範圍
  112: ATR = ta.atr(atrLength)
  113: 
  114: // ? 固定 0.5% 止盈計算
  115: takeProfitPercent = 0.007  // 0.7% 固定止盈
  116: takeProfitLong = close * (1 + takeProfitPercent)  // 多單止盈
  117: takeProfitShort = close * (1 - takeProfitPercent) // 空單止盈

## Exits every bar  [lines 145-149]
  145: stopLossLong = close - (ATR * atrMultiplier)  // 多單 ATR 止損

  148: strategy.exit("Exit Long", from_entry="B", stop=stopLossLong, limit=takeProfitLong, comment="TP Long")
