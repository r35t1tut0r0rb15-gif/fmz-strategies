## Stop 1 % of the average price (off)  [lines 135-163]
  135: enableStopLoss = input.bool(false, title="Enable Stop Loss")
  136: stopLossPercent = input.float(1.0, title="Stop Loss Percentage (%)", minval=0.1) / 100

  138: // Bollinger Bands calculation
  139: basis = ta.sma(src, length)
  140: dev = mult * ta.stdev(src, length)
  141: upperBB = basis + dev
  142: lowerBB = basis - dev
  143: 
  144: // RSI calculation
  145: rsi = ta.rsi(src, rsiLength)
  146: 
  147: // 200 EMA to filter trades (calculated but only used if enabled)
  148: ema200 = ta.ema(src, emaLength)
  149: 
  150: // Long condition: RSI below oversold, price closes below the lower Bollinger Band, and optionally price is above the 200 EMA
  151: longCondition = enableLong and (rsi < rsiOversold) and (close < lowerBB) and (not enableEMAFilter or close > ema200)
  152: if (longCondition)
  153:     strategy.entry("Long", strategy.long)
  154: 
  155: // Short condition: RSI above overbought, price closes above the upper Bollinger Band, and optionally price is below the 200 EMA
  156: shortCondition = enableShort and (rsi > rsiOverbought) and (close > upperBB) and (not enableEMAFilter or close < ema200)
  157: if (shortCondition)
  158:     strategy.entry("Short", strategy.short)
  159: 
  160: // Stop Loss setup
  161: if (enableStopLoss)
  162:     strategy.exit("Long Exit", "Long", stop = strategy.position_avg_price * (1 - stopLossPercent))
  163:     strategy.exit("Short Exit", "Short", stop = strategy.position_avg_price * (1 + stopLossPercent))
