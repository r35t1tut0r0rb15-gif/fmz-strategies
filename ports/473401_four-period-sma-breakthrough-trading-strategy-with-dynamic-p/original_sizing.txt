  111: // Calculate Take Profit and Stop Loss
  112: longTakeProfit = strategy.position_avg_price * (1 + takeProfitPercent / 100)  // TP for long

  114: 
  115: shortTakeProfit = strategy.position_avg_price * (1 - takeProfitPercent / 100) // TP for short
  116: shortStopLoss = strategy.position_avg_price * (1 + stopLossPercent / 100)     // SL for short
  117: 
  118: // Exit for Long
  119: if (strategy.position_size > 0)  // If in a long position
  120:     strategy.exit("Long TP/SL", from_entry="Long", limit=longTakeProfit, stop=longStopLoss)
  121: 
  122: // Exit for Short
  123: if (strategy.position_size < 0)  // If in a short position
  124:     strategy.exit("Short TP/SL", from_entry="Short", limit=shortTakeProfit, stop=shortStopLoss)
