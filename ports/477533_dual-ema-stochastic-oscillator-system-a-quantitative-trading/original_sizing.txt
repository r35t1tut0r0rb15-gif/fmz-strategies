## Stop 1 %, risk-reward 2  [lines 105-141]
  105: riskRewardRatio = input.float(2.0, title="Risk-Reward Ratio")
  106: stopLossPercent = input.float(1.0, title="Stop Loss (%)")
  107: 
  108: // EMA Calculation
  109: emaShort = ta.ema(close, emaShortLength)
  110: emaLong = ta.ema(close, emaLongLength)
  111: 
  112: // Stochastic Calculation
  113: k = ta.stoch(high, low, close, stochK)
  114: d = ta.sma(k, stochD)
  115: 
  116: // Trend Condition
  117: isUptrend = emaShort > emaLong
  118: isDowntrend = emaShort < emaLong
  119: 
  120: // Stochastic Signals
  121: stochBuyCrossover = ta.crossover(k, d)
  122: stochBuySignal = k < stochOversold and stochBuyCrossover
  123: stochSellCrossunder = ta.crossunder(k, d)
  124: stochSellSignal = k > stochOverbought and stochSellCrossunder
  125: 
  126: // Entry Signals
  127: buySignal = isUptrend and stochBuySignal
  128: sellSignal = isDowntrend and stochSellSignal
  129: 
  130: // Strategy Execution
  131: if buySignal
  132:     strategy.entry("Buy", strategy.long)
  133:     stopLoss = close * (1 - stopLossPercent / 100)
  134:     takeProfit = close * (1 + stopLossPercent * riskRewardRatio / 100)
  135:     strategy.exit("Take Profit/Stop Loss", from_entry="Buy", stop=stopLoss, limit=takeProfit)
  136: 
  137: if sellSignal
  138:     strategy.entry("Sell", strategy.short)
  139:     stopLoss = close * (1 + stopLossPercent / 100)
  140:     takeProfit = close * (1 - stopLossPercent * riskRewardRatio / 100)
  141:     strategy.exit("Take Profit/Stop Loss", from_entry="Sell", stop=stopLoss, limit=takeProfit)
