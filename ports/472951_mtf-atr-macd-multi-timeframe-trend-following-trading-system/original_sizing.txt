## Trailing stop 1.5 ATR / target 2 ATR (sent on reversal bars)  [lines 128-182]
  128: atrPeriod = input.int(14, title="ATR Period", minval=1)
  129: atrMultiplier = input.float(1.5, title="ATR Multiplier", minval=0.1)

  132: // Multi-timeframe settings

  138: macdLongPeriod = input.int(26, title="MACD Long Period", minval=1)
  139: macdSignalPeriod = input.int(9, title="MACD Signal Period", minval=1)

  144: // Calculating indicators
  145: emaShort = ta.ema(close, emaShortPeriod)
  146: emaLong = ta.ema(close, emaLongPeriod)
  147: rsiValue = ta.rsi(close, rsiPeriod)
  148: atrValue = ta.atr(atrPeriod)
  149: 
  150: // Calculate MACD
  151: [macdLine, macdSignalLine, _] = ta.macd(close, macdShortPeriod, macdLongPeriod, macdSignalPeriod)
  152: 
  153: // Higher timeframe EMA confirmation
  154: htfEMALong = request.security(syminfo.tickerid, htfEMATimeframe, ta.ema(close, emaLongPeriod))
  155: 
  156: // Trading conditions
  157: longCondition = ta.crossover(emaShort, emaLong) and rsiValue < rsiOverbought and (not htfEMAEnabled or close > htfEMALong) and macdLine > macdSignalLine
  158: shortCondition = ta.crossunder(emaShort, emaLong) and rsiValue > rsiOversold and (not htfEMAEnabled or close < htfEMALong) and macdLine < macdSignalLine
  159: 
  160: // Plotting EMAs
  161: plot(emaShort, title="EMA Short", color=color.green)
  162: plot(emaLong, title="EMA Long", color=color.red)

  165: hline(0, "Zero Line", color=color.gray)
  166: plot(macdLine - macdSignalLine, title="MACD Histogram", color=color.green, style=plot.style_histogram)
  167: plot(macdLine, title="MACD Line", color=color.blue)
  168: plot(macdSignalLine, title="MACD Signal Line", color=color.red)
  169: 
  170: // Trailing Stop-Loss and Take-Profit levels
  171: var float trailStopLoss = na
  172: var float trailTakeProfit = na
  173: 
  174: if (strategy.position_size > 0) // Long Position
  175:     trailStopLoss := na(trailStopLoss) ? close - atrValue * atrMultiplier : math.max(trailStopLoss, close - atrValue * atrMultiplier)
  176:     trailTakeProfit := close + atrValue * takeProfitATRMultiplier
  177:     strategy.exit("Exit Long", "Long", stop=trailStopLoss, limit=trailTakeProfit, when=shortCondition)
  178: 
  179: if (strategy.position_size < 0) // Short Position
  180:     trailStopLoss := na(trailStopLoss) ? close + atrValue * atrMultiplier : math.min(trailStopLoss, close + atrValue * atrMultiplier)
  181:     trailTakeProfit := close - atrValue * takeProfitATRMultiplier
  182:     strategy.exit("Exit Short", "Short", stop=trailStopLoss, limit=trailTakeProfit, when=longCondition)
