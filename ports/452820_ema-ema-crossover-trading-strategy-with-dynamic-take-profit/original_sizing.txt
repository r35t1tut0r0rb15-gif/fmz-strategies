## Target 8 % / stop 2.5 % of the average price  [lines 126-156]
  126: takeProfitPercent = 0.080
  127: stopLossPercent = 0.025
  128: 
  129: // คำนวณระดับ Take Profit และ Stop Loss
  130: longTakeProfit = strategy.position_avg_price * (1 + takeProfitPercent)
  131: longStopLoss = strategy.position_avg_price * (1 - stopLossPercent)
  132: 
  133: shortTakeProfit = strategy.position_avg_price * (1 - takeProfitPercent)
  134: shortStopLoss = strategy.position_avg_price * (1 + stopLossPercent)
  135: 
  136: // สัญญาณ Buy
  137: buySignal = (ema12 > ema26) and (ema12[1] <= ema26[1])
  138: 
  139: // สัญญาณ Sell
  140: sellSignal = (ema12 < ema26) and (ema12[1] >= ema26[1])
  141: 
  142: // เปิด Position Long
  143: if (buySignal)
  144:     strategy.entry("Long", strategy.long)
  145: 
  146: // เปิด Position Short
  147: if (sellSignal)
  148:     strategy.entry("Short", strategy.short)
  149: 
  150: // ปิด Position Long เมื่อถึง Take Profit หรือ Stop Loss
  151: if (strategy.position_size > 0)
  152:     strategy.exit("Long TP/SL", from_entry="Long", limit=longTakeProfit, stop=longStopLoss, comment="TP/SL")
  153: 
  154: // ปิด Position Short เมื่อถึง Take Profit หรือ Stop Loss
  155: if (strategy.position_size < 0)
  156:     strategy.exit("Short TP/SL", from_entry="Short", limit=shortTakeProfit, stop=shortStopLoss, comment="TP/SL")
