  337: longStopPrice := if (strategy.position_size > 0)
  338:     stopValue = close * (1 - TrailPerc)

  343: shortStopPrice=0.0//short side entry stop variable
  344: shortStopPrice := if (strategy.position_size < 0)
  345:     shortStopValue = close * (1 + TrailPerc)

  347: else
  348:     999999 
  349:

  351: // Determine where you've entered and in what direction
  352: longStop = strategy.position_avg_price * (1 - stopPer)
  353: shortStop = strategy.position_avg_price * (1 + stopPer)
  354: shortTake = strategy.position_avg_price * (1 - takePer)
  355: longTake = strategy.position_avg_price * (1 + takePer)
