  110: 
  111: // Bollinger Bands calculation
  112: basis = ta.sma(close, length)
  113: upperBand = basis + mult * ta.stdev(close, length)
  114: lowerBand = basis - mult * ta.stdev(close, length)
  115: 
  116: // Conditions for Open Long
  117: openLongCondition = strategy.position_size == 0 and close > open and (close + open) / 2 > upperBand
  118: 
  119: // Conditions for Open Short
  120: openShortCondition = strategy.position_size == 0 and close < open and (close + open) / 2 < lowerBand
  121: 
  122: // Conditions for Close Long
  123: closeLongCondition = strategy.position_size > 0 and strategy.position_size > 0 and (close < upperBand and open < upperBand and close < open)
  124: 
  125: // Conditions for Close Short
  126: closeShortCondition = strategy.position_size < 0 and strategy.position_size < 0 and (close > lowerBand and open > lowerBand and close > open)
  127: 
  128: 
  129: // Long entry
  130: if openLongCondition
  131:     strategy.entry("Long", strategy.long, qty=strategy.equity * lev / close)
  132:     strategy.exit("Long SL", from_entry="Long", stop=low)  // Set Stop-Loss
