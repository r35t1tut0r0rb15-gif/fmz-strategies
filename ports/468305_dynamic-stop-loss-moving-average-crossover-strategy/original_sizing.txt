## Stop at the 5-bar swing, target 3 x the risk; trail_offset 0  [lines 269-299]
  269: trailOffset = input(0, title="Trailing Drawdown Offset")
  270: 
  271: // Calculate moving averages
  272: shortEMA = ta.ema(close, shortMA)
  273: longEMA = ta.ema(close, longMA)
  274: 
  275: // Plot moving averages
  276: plot(shortEMA, color=color.blue, title="Short EMA")
  277: plot(longEMA, color=color.red, title="Long EMA")
  278: 
  279: // Identify recent swing high and low
  280: swingHigh = ta.highest(high, 5)
  281: swingLow = ta.lowest(low, 5)
  282: 
  283: // Buy condition: EMA crossover
  284: longCondition = ta.crossover(shortEMA, longEMA)
  285: if (longCondition)
  286:     strategy.close("Short")  // Close any existing short position
  287:     stopLoss = swingLow  // At swing low
  288:     takeProfit = close + (3 * (close - stopLoss))  // 1:3 RR
  289:     strategy.entry("Long", strategy.long)
  290:     strategy.exit("TP/SL", "Long", limit=takeProfit, stop=stopLoss, trail_offset=trailOffset)
  291: 
  292: // Sell condition: EMA crossover
  293: shortCondition = ta.crossunder(shortEMA, longEMA)
  294: if (shortCondition)
  295:     strategy.close("Long")  // Close any existing long position
  296:     stopLoss = swingHigh  // At swing high
  297:     takeProfit = close - (3 * (stopLoss - close))  // 1:3 RR
  298:     strategy.entry("Short", strategy.short)
  299:     strategy.exit("TP/SL", "Short", limit=takeProfit, stop=stopLoss, trail_offset=trailOffset)
