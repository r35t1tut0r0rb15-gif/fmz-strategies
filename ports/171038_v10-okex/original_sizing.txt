  102:    var account = _C(exchange.GetAccount);
  103:   
  104:    for (var i = 0; i < orders.length; i++) {
  105:        var order=orders[i];
  106:        if(order.type==1){  //空单
  107:            exchange.CancelOrder(order.Id);
  108:            Log("清空单");
  109:        }
  110:    }
  111:    
  112:    var ticker = _C(exchange.GetTicker);

  116:    var pAmount=0;
  117:    for (i = 0; i < position.length; i++) {
  118:        if(position[i].Type==PD_SHORT){ //买平空单,注意成交方向相反
  119:            exchange.SetDirection("closesell"); 
  120:            exchange.Buy(nowPrice,position[i].Amount);
  121:        }else{
  122:            pAmount=position[i].Amount;
  123:        }

  125:
