## Stop 1.5 % of the avg price (danger-armed)  [lines 119-155]
  119: stop_loss_pct = input.float(1.5, title="止损百分比", minval=0.1)
  120: 
  121: // === 指标计算 ===
  122: // 均线
  123: ema_short = ta.ema(close, ema_short_length)
  124: ema_long = ta.ema(close, ema_long_length)
  125: 
  126: // MACD
  127: [macd_line, signal_line, _] = ta.macd(close, macd_fast_length, macd_slow_length, macd_signal_length)
  128: 
  129: // RSI
  130: rsi = ta.rsi(close, rsi_length)
  131: 
  132: // === 趋势过滤 ===
  133: bullish_trend = ema_short > ema_long  // 多头趋势：短期均线高于长期均线
  134: bearish_trend = ema_short < ema_long  // 空头趋势：短期均线低于长期均线
  135: 
  136: // === 买入条件 ===
  137: // 1. EMA 显示多头趋势
  138: // 2. MACD 金叉（MACD 线向上突破信号线）
  139: // 3. RSI 不在超买区域
  140: buy_signal = bullish_trend and ta.crossover(macd_line, signal_line) and rsi < rsi_overbought
  141: 
  142: // === 危险信号（动态止盈条件） ===
  143: // 1. 趋势反转：短期均线跌破长期均线，或者价格跌破短期均线
  144: // 2. MACD 死叉：MACD 线向下跌破信号线
  145: // 3. RSI：RSI 超买并开始回落
  146: danger_signal = bearish_trend or close < ema_short or ta.crossunder(macd_line, signal_line) or (rsi > rsi_overbought and ta.falling(rsi, 2))  // 检查 RSI 最近2周期是否下降
  147: 
  148: // === 策略执行 ===
  149: if (buy_signal)
  150:     strategy.entry("Buy", strategy.long)
  151: 
  152: // 动态止盈和止损
  153: if (strategy.position_size > 0)
  154:     stop_price = strategy.position_avg_price * (1 - stop_loss_pct / 100)  // 固定止损
  155:     strategy.exit("Exit", from_entry="Buy", stop=stop_price, when=danger_signal)
