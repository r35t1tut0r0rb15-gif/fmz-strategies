  376: risk = input(2,type=input.float,title="Risk %")/100           //risk % per trade

  387:     //Calculate current DD and determine if stopout is necessary

  390: if(floating<0 and abs(floating/balance)>equity_protector)
  391:     equity_stopout := true
  392:     
  393: equity_stopout2 = false
  394: if(floating>0 and abs(floating/balance)>equity_protectorTP)
  395:     equity_stopout2 := true
