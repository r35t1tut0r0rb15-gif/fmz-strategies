   99: // —————— Inputs ——————
  100: emaFastLength = input.int(20, "Fast EMA")
  101: emaSlowLength = input.int(50, "Slow EMA")
  102: rsiLength = input.int(14, "RSI Length")
  103: bbLength = input.int(20, "Bollinger Bands Length")
  104: riskRewardRatio = input.float(1.5, "Risk/Reward Ratio")
  105: stopLossPerc = input.float(2, "Stop Loss %") / 100

  107: // —————— Indicators ——————
  108: // Trend: EMAs
  109: emaFast = ta.ema(close, emaFastLength)
  110: emaSlow = ta.ema(close, emaSlowLength)
  111: ema200 = ta.ema(close, 200)
  112: 
  113: // Momentum: RSI & MACD
  114: rsi = ta.rsi(close, rsiLength)
  115: [macdLine, signalLine, _] = ta.macd(close, 12, 26, 9)
  116: 
  117: // Volatility: Bollinger Bands
  118: basis = ta.sma(close, bbLength)
  119: dev = ta.stdev(close, bbLength)
  120: upperBand = basis + 2 * dev
  121: lowerBand = basis - 2 * dev
  122: 
  123: // —————— Strategy Logic ——————
  124: // Long Conditions
  125: longCondition = 
  126:   close > ema200 and // Long-term bullish
  127:   ta.crossover(emaFast, emaSlow) and // EMA crossover
  128:   rsi > 50 and // Momentum rising
  129:   close > lowerBand and // Bounce from lower Bollinger Band
  130:   macdLine > signalLine // MACD bullish
  131: 
  132: // Short Conditions
  133: shortCondition = 
  134:   close < ema200 and // Long-term bearish
  135:   ta.crossunder(emaFast, emaSlow) and // EMA crossunder
  136:   rsi < 50 and // Momentum weakening
  137:   close < upperBand and // Rejection from upper Bollinger Band
  138:   macdLine < signalLine // MACD bearish
  139: 
  140: // —————— Risk Management ——————
  141: stopLoss = strategy.position_avg_price * (1 - stopLossPerc)
  142: takeProfit = strategy.position_avg_price * (1 + (riskRewardRatio * stopLossPerc))
  143: 
  144: // —————— Execute Trades ——————
  145: if (longCondition)
  146:     strategy.entry("Long", strategy.long)
  147:     strategy.exit("Exit Long", "Long", stop=stopLoss, limit=takeProfit)
  148: 
  149: if (shortCondition)
  150:     strategy.entry("Short", strategy.short)
  151:     strategy.exit("Exit Short", "Short", stop=stopLoss, limit=takeProfit)
