## qty = 3 % of 10000 / (1.5 ATR); 100 % of equity  [lines 157-228]
  157: strategy("Smart Trade By Amit Roy", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=100)
  158: 
  159: // Input Settings
  160: riskPercent = input.float(3, title="Risk Percentage per Trade (%)", minval=0.1, step=0.1)
  161: rewardRatio = input.float(3, title="Risk-Reward Ratio", minval=1.0)
  162: capital = input.float(10000, title="Starting Capital ($)", minval=1)
  163: atrMultiplier = input.float(1.5, title="ATR Multiplier for Stop Loss")
  164: show_TT = input.bool(true, title = "Show T and TT")
  165: show_sma = input.bool(true, title = "Show SMA")
  166: 
  167: // ATR Calculation for Volatility-based Stop-Loss
  168: atrLength = input.int(14, title="ATR Length")
  169: atrValue = ta.atr(atrLength)
  170: stopLossDistance = atrValue * atrMultiplier
  171: takeProfitDistance = stopLossDistance * rewardRatio
  172: 
  173: // Position Sizing Calculation
  174: riskAmount = capital * (riskPercent / 100)
  175: positionSize = riskAmount / stopLossDistance
  176: 
  177: // Simple Moving Averages
  178: fastMA = ta.sma(close, 13)
  179: slowMA = ta.sma(close, 5)
  180: 
  181: // Entry and Exit Conditions using Simple Moving Averages
  182: longCondition = ta.crossover(fastMA, slowMA)
  183: shortCondition = ta.crossunder(fastMA, slowMA)
  184: 
  185: // Candlestick Patterns Functions
  186: isBullishEngulfing() => (open[1] > close[1] and close > open and close >= open[1] and close[1] >= open and close - open > open[1] - close[1])
  187: isBearishEngulfing() => (close[1] > open[1] and open > close and open >= close[1] and open[1] >= close and open - close > close[1] - open[1])
  188: isHammer() => (((high - low) > 3 * (open - close)) and ((close - low) / (.001 + high - low) > 0.6) and ((open - low) / (.001 + high - low) > 0.6))
  189: isInvertedHammer() => (((high - low) > 3 * (open - close)) and ((high - close) / (.001 + high - low) > 0.6) and ((high - open) / (.001 + high - low) > 0.6))
  190: isBullishHarami() => (open[1] > close[1] and close > open and close <= open[1] and close[1] <= open and close - open < open[1] - close[1])
  191: isBearishHarami() => (close[1] > open[1] and open > close and open <= close[1] and open[1] <= close and open - close < close[1] - open[1])
  192: 
  193: // Color Bars for Candlestick Patterns
  194: barcolor(isBullishEngulfing() ? color.rgb(0, 102, 255) : na)
  195: barcolor(isHammer() ? (#1f0cef) : na)
  196: barcolor(isBullishHarami() ? color.rgb(0, 93, 214) : na)
  197: barcolor(isBearishEngulfing() ? color.rgb(255, 196, 0) : na)
  198: barcolor(isBearishHarami() ? color.rgb(251, 255, 0) : na)
  199: barcolor(isInvertedHammer() ? color.rgb(247, 0, 247) : na)
  200: 
  201: // Calculate SMA for Visualization
  202: sma_22 = ta.sma(close, 22)
  203: lineColor = close > sma_22 ? color.green : color.green
  204: plot(show_sma ? sma_22 : na, color=lineColor, linewidth=1)
  205: 
  206: // Determine T and TT Labels based on Conditions
  207: candleCrossG = ta.crossover(close, sma_22)
  208: candleCrossR = ta.crossunder(close, sma_22)
  209: 
  210: // Plot T and TT labels
  211: redT = candleCrossG and close < open
  212: greenTT = candleCrossG and close > open and close > sma_22
  213: greenT = candleCrossR and close > open
  214: redTT = candleCrossR and close < open
  215: 
  216: plotshape(series=redT ? show_TT : na, title="Red-T", color=na, style=shape.labeldown, location=location.abovebar, size=size.small, textcolor=color.red, text="T")
  217: plotshape(series=greenTT ? show_TT : na, title="Green-TT", color=na, style=shape.labelup, location=location.belowbar, size=size.tiny, textcolor=color.green, text="TT")
  218: plotshape(series=greenT ? show_TT : na, title="Green-T", color=na, style=shape.labelup, location=location.belowbar, size=size.small, textcolor=color.green, text="T")
  219: plotshape(series=redTT ? show_TT : na, title="Red-TT", color=na, style=shape.labeldown, location=location.abovebar, size=size.tiny, textcolor=color.red, text="TT")
  220: 
  221: // Place Trades Based on Conditions
  222: if (longCondition)
  223:     strategy.entry("उड़ाओ ", strategy.long, qty=positionSize)
  224:     strategy.exit("Take Profit", from_entry="Long", limit=close + takeProfitDistance, stop=close - stopLossDistance)
  225: 
  226: if (shortCondition)
  227:     strategy.entry("गिराओ", strategy.short, qty=positionSize)
  228:     strategy.exit("Take Profit", from_entry="Short", limit=close - takeProfitDistance, stop=close + stopLossDistance)
