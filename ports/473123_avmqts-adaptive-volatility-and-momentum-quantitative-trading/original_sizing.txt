## Trade size 5 ETH; target 9 % / stop 9.9 % of the average price  [lines 88-154]
   88: trade_size = input.float(5, title="Trade Size (ETH)")
   89: atr_length = input.int(8, minval=1, title="ATR Length")
   90: macd_fast = input.int(8, minval=1, title="MACD Fast Length")
   91: macd_slow = input.int(7, minval=1, title="MACD Slow Length")
   92: macd_signal = input.int(9, minval=1, title="MACD Signal Length")
   93: momentum_length = input.int(37, title="Momentum Length")
   94: stop_loss_percent = input.float(9.9, title="Stop Loss Percentage (%)")
   95: take_profit_percent = input.float(9.0, title="Take Profit Percentage (%)")
   96: alternate_signal = input.bool(true, title="Alternate Buy/Sell Signals")
   97: 
   98: // === Indicators === //
   99: // ATR to measure volatility
  100: atr = ta.atr(atr_length)
  101: 
  102: // MACD for trend momentum
  103: [macd_line, signal_line, _] = ta.macd(close, macd_fast, macd_slow, macd_signal)
  104: macd_cross_up = ta.crossover(macd_line, signal_line)
  105: macd_cross_down = ta.crossunder(macd_line, signal_line)
  106: 
  107: // Momentum
  108: momentum = ta.mom(close, momentum_length)
  109: 
  110: // === Signal Control Variables === //
  111: var bool last_signal_long = na
  112: var int last_trade_bar = na
  113: min_bars_between_trades = 5 // Adjust for minimal trade frequency control
  114: time_elapsed = na(last_trade_bar) or (bar_index - last_trade_bar) >= min_bars_between_trades
  115: 
  116: // === Buy and Sell Conditions === //
  117: // Buy when:
  118: buy_signal = (macd_cross_up and momentum > 0 and close > ta.sma(close, 50) and time_elapsed)
  119: 
  120: // Sell when:
  121: sell_signal = (macd_cross_down and momentum < 0 and close < ta.sma(close, 50) and time_elapsed)
  122: 
  123: // Enforce alternate signals if selected
  124: if alternate_signal
  125:     buy_signal := buy_signal and (na(last_signal_long) or not last_signal_long)
  126:     sell_signal := sell_signal and (not na(last_signal_long) and last_signal_long)
  127: 
  128: // === Trade Execution === //
  129: // Buy Position
  130: if (buy_signal)
  131:     if strategy.position_size < 0
  132:         strategy.close("Short")
  133:     strategy.entry("Long", strategy.long, qty=trade_size)
  134:     last_signal_long := true
  135:     last_trade_bar := bar_index
  136: 
  137: // Sell Position
  138: if (sell_signal)
  139:     if strategy.position_size > 0
  140:         strategy.close("Long")
  141:     strategy.entry("Short", strategy.short, qty=trade_size)
  142:     last_signal_long := false
  143:     last_trade_bar := bar_index
  144: 
  145: // === Stop Loss and Take Profit === //
  146: if strategy.position_size > 0
  147:     long_take_profit = strategy.position_avg_price * (1 + take_profit_percent / 100)
  148:     long_stop_loss = strategy.position_avg_price * (1 - stop_loss_percent / 100)
  149:     strategy.exit("TP/SL Long", from_entry="Long", limit=long_take_profit, stop=long_stop_loss)
  150: 
  151: if strategy.position_size < 0
  152:     short_take_profit = strategy.position_avg_price * (1 - take_profit_percent / 100)
  153:     short_stop_loss = strategy.position_avg_price * (1 + stop_loss_percent / 100)
  154:     strategy.exit("TP/SL Short", from_entry="Short", limit=short_take_profit, stop=short_stop_loss)
