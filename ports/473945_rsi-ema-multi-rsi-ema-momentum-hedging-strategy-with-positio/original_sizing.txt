## strategy(): fixed qty 1; qty = equity x 20 / close, doubled on repeat entries (pyramiding 1)  [lines 101-173]
  101: strategy("Custom RSI EMA Strategy Hedge", overlay=true, default_qty_type=strategy.fixed, default_qty_value=1)
  102: 
  103: // Definování vstupních podmínek
  104: rsi_14 = ta.rsi(close, 14)
  105: rsi_2 = ta.rsi(close, 2)
  106: ema_50 = ta.ema(close, 50)
  107: ema_100 = ta.ema(close, 100)
  108: ema_200 = ta.ema(close, 200)
  109: 
  110: // Pákový efekt
  111: leverage = 20
  112: 
  113: // Podmínky pro long pozici
  114: longCondition = (rsi_14[1] < 31) and ta.crossover(rsi_2, 10) and (ema_50 < ema_100) and (ema_100 < ema_200)
  115: 
  116: // Podmínky pro short pozici
  117: shortCondition = (rsi_14[1] > 69) and ta.crossunder(rsi_2, 90) and (ema_50 > ema_100) and (ema_100 > ema_200)
  118: 
  119: // Identifikátory pozic
  120: long_id = "Long"
  121: short_id = "Short"
  122: 
  123: // Definování průměrné ceny pozice pro long a short
  124: var float long_avg_price = na
  125: var float short_avg_price = na
  126: 
  127: // Sledujeme, zda se velikost pozice změnila
  128: var float last_long_position_size = na
  129: var float last_short_position_size = na
  130: 
  131: // Přerušení průměrné ceny pozice při změně pozice
  132: if (last_long_position_size != strategy.position_size and strategy.position_size > 0)
  133:     long_avg_price := na
  134: if (last_short_position_size != strategy.position_size and strategy.position_size < 0)
  135:     short_avg_price := na
  136: 
  137: // Aktualizace průměrné ceny pozice
  138: if (strategy.position_size > 0)
  139:     long_avg_price := strategy.position_avg_price
  140: else
  141:     long_avg_price := na
  142: 
  143: if (strategy.position_size < 0)
  144:     short_avg_price := strategy.position_avg_price
  145: else
  146:     short_avg_price := na
  147: 
  148: // Uložení aktuální velikosti pozice pro příští bar
  149: if (strategy.position_size > 0)
  150:     last_long_position_size := strategy.position_size
  151: else if (strategy.position_size < 0)
  152:     last_short_position_size := strategy.position_size
  153: 
  154: // Podmínky pro take profit
  155: takeProfitLongCondition = (rsi_14 > 69) and (rsi_2 > 90) and (long_avg_price < close)
  156: takeProfitShortCondition = (rsi_14 < 31) and (rsi_2 < 10) and (short_avg_price > close)
  157: 
  158: // Velikost pozice
  159: new_long_position_size = strategy.position_size == 0 ? na : math.abs(strategy.position_size) * 2
  160: new_short_position_size = strategy.position_size == 0 ? na : math.abs(strategy.position_size) * 2
  161: 
  162: // Úprava velikosti pozice s ohledem na pákový efekt
  163: position_value = strategy.equity * leverage
  164: trade_qty_long = position_value / close
  165: trade_qty_short = position_value / close
  166: 
  167: // Vstup do long pozice s dvojnásobkem aktuální pozice nebo standardní velikostí při první pozici
  168: if (longCondition)
  169:     strategy.entry(long_id, strategy.long, qty=new_long_position_size == na ? trade_qty_long : new_long_position_size)
  170: 
  171: // Vstup do short pozice s dvojnásobkem aktuální pozice nebo standardní velikostí při první pozici
  172: if (shortCondition)
  173:     strategy.entry(short_id, strategy.short, qty=new_short_position_size == na ? trade_qty_short : new_short_position_size)
