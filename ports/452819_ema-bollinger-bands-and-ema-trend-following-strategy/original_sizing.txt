## Stop 0.5 % / target 1 % of the signal close; risk 1 % of equity over the stop distance  [lines 135-159]
  135: stop_loss_pct = input.float(0.5, title="Stop Loss (%)")
  136: take_profit_pct = input.float(1.0, title="Take Profit (%)")
  137: stop_loss_level_long = close * (1 - stop_loss_pct / 100)
  138: take_profit_level_long = close * (1 + take_profit_pct / 100)
  139: stop_loss_level_short = close * (1 + stop_loss_pct / 100)
  140: take_profit_level_short = close * (1 - take_profit_pct / 100)
  141: 
  142: // Calculate Position Size Based on Risk Per Trade
  143: risk_per_trade = input.float(1.0, title="Risk Per Trade (%)")
  144: capital_at_risk = strategy.equity * risk_per_trade / 100
  145: risk_per_unit_long = math.abs(close - stop_loss_level_long)
  146: risk_per_unit_short = math.abs(close - stop_loss_level_short)
  147: position_size_long = capital_at_risk / risk_per_unit_long
  148: position_size_short = capital_at_risk / risk_per_unit_short
  149: 
  150: // Enter Long and Short Trades
  151: if long_condition
  152:     strategy.entry("Long", strategy.long, qty=position_size_long)
  153:     strategy.exit("Take Profit", "Long", limit=take_profit_level_long)
  154:     strategy.exit("Stop Loss", "Long", stop=stop_loss_level_long)
  155: 
  156: if short_condition
  157:     strategy.entry("Short", strategy.short, qty=position_size_short)
  158:     strategy.exit("Take Profit", "Short", limit=take_profit_level_short)
  159:     strategy.exit("Stop Loss", "Short", stop=stop_loss_level_short)
