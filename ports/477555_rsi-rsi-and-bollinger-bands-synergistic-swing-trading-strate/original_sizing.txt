  137: bbLength = input.int(20, minval=1, title="Bollinger Bands Length")
  138: bbMult = input.float(2.0, minval=0.1, maxval=5, title="Bollinger Bands StdDev Multiplier")
  139: maLength = input.int(50, minval=1, title="Moving Average Length")

  142: rsi = ta.rsi(close, rsiLength)
  143: 
  144: // Bollinger Bands Calculation
  145: bbBasis = ta.sma(close, bbLength)
  146: bbDev = bbMult * ta.stdev(close, bbLength)
  147: bbUpper = bbBasis + bbDev
  148: bbLower = bbBasis - bbDev
  149: 
  150: // Moving Average
  151: ma = ta.sma(close, maLength)
  152: 
  153: // Buy Signal: Price near or below lower Bollinger Band AND RSI below oversold level
  154: buySignal = (close <= bbLower * 1.01) and (rsi < rsiOversold)
  155: 
  156: // Sell Signal: Price near or above upper Bollinger Band OR RSI above overbought level
  157: sellSignal = (close >= bbUpper * 0.99) or (rsi > rsiOverbought)
  158: 
  159: // Date Range Inputs
  160: startDate = input(timestamp("2018-01-01 00:00"), title="Start Date")

  162: inDateRange = true
