## 5 contracts; stop 1 ATR / target 1.5 ATR  [lines 105-138]
  105: rrRatio = input.float(1.5, "Risk-Reward Ratio")
  106: contracts = input.int(5, "Number of Contracts")
  107: 
  108: // Calculations
  109: ema20 = ta.ema(close, emaShort)

  113: atr = ta.atr(14)
  114: 
  115: // Conditions
  116: longCondition = ta.crossover(close, ema20) and close > ema50
  117: shortCondition = ta.crossunder(close, ema20) and close < ema50
  118: 
  119: // Variables for trades
  120: var float entryPrice = na
  121: var float stopLoss = na
  122: var float takeProfit = na
  123: 
  124: // Long Trades
  125: if (longCondition)
  126:     entryPrice := close
  127:     stopLoss := close - atr
  128:     takeProfit := close + atr * rrRatio

  130:     strategy.exit("Exit Long", from_entry="Long", stop=stopLoss, limit=takeProfit)

  132: // Short Trades

  134:     entryPrice := close
