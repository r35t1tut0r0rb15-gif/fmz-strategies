## Stop / target 1 % of the current close (every bar)  [lines 103-130]
  103: sl_pct = input(1, title="Stop-Loss (%)") / 100
  104: tp_pct = input(1, title="Take-Profit (%)") / 100
  105: 
  106: // Belirtilen Zaman Diliminde RSI ve EMA Hesaplamaları
  107: rsi = ta.rsi(close, rsi_period)
  108: ema = ta.ema(close, ema_period)
  109: 
  110: // Long ve Short Sinyalleri
  111: long_signal = rsi[2] > rsi_overbought and rsi < rsi_overbought  and (close > ema or not use_ema)
  112: short_signal = rsi[2] < rsi_oversold and rsi > rsi_oversold and (close < ema or not use_ema)
  113: 
  114: // Alım/Satım İşlemleri
  115: if long_signal
  116:     strategy.entry("Long", strategy.long)
  117: 
  118: if short_signal
  119:     strategy.entry("Short", strategy.short)
  120: 
  121: // Stop-Loss ve Take-Profit Uygulaması
  122: if strategy.position_size > 0
  123:     long_stop_loss = close * (1 - sl_pct)
  124:     long_take_profit = close * (1 + tp_pct)
  125:     strategy.exit("Long Exit", from_entry="Long", stop=long_stop_loss, limit=long_take_profit)
  126: 
  127: if strategy.position_size < 0
  128:     short_stop_loss = close * (1 + sl_pct)
  129:     short_take_profit = close * (1 - tp_pct)
  130:     strategy.exit("Short Exit", from_entry="Short", stop=short_stop_loss, limit=short_take_profit)
