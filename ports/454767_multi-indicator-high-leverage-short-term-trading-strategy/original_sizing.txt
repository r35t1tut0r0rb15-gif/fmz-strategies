## Stop / target 1.5 ATR 5 from the signal close  [lines 184-212]
  184: atrLength = input.int(5, minval=1, title="ATR Length")
  185: atrMultiplier = input.float(1.5, minval=0.1, title="ATR Multiplier")
  186: 
  187: // Indicators
  188: shortEma = ta.ema(close, shortEmaLength)
  189: longEma = ta.ema(close, longEmaLength)
  190: rsi = ta.rsi(close, rsiLength)
  191: [macdLine, signalLine, _] = ta.macd(close, macdFastLength, macdSlowLength, macdSignalSmoothing)
  192: atr = ta.atr(atrLength)
  193: 
  194: // Conditions
  195: longCondition = ta.crossover(shortEma, longEma) and rsi < rsiOverbought and macdLine > signalLine
  196: shortCondition = ta.crossunder(shortEma, longEma) and rsi > rsiOversold and macdLine < signalLine
  197: 
  198: // Dynamic stop-loss and take-profit levels
  199: longStopLoss = close - (atr * atrMultiplier)
  200: longTakeProfit = close + (atr * atrMultiplier)
  201: shortStopLoss = close + (atr * atrMultiplier)
  202: shortTakeProfit = close - (atr * atrMultiplier)
  203: 
  204: // Long Entry
  205: if (longCondition)
  206:     strategy.entry("Long", strategy.long)
  207:     strategy.exit("Take Profit/Stop Loss", from_entry="Long", limit=longTakeProfit, stop=longStopLoss)
  208: 
  209: // Short Entry
  210: if (shortCondition)
  211:     strategy.entry("Short", strategy.short)
  212:     strategy.exit("Take Profit/Stop Loss", from_entry="Short", limit=shortTakeProfit, stop=shortStopLoss)
