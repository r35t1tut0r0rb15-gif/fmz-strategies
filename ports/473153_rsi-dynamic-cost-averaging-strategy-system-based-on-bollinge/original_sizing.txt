  149: dev = mult * ta.stdev(close, length)
  150: upper = basis + dev

  164: sellSignal = (rsi > rsiSellLevel and close >= upper)

  166: // DCA Strategy: Enter Long or Short based on signals with calculated position size

  170: if (sellSignal)

  173: // Take Profit Logic
  174: if (strategy.position_size > 0)  // If long
  175:     strategy.exit("Take Profit Long", from_entry="DCA Buy", limit=close * (1 + takeProfitPercentage / 100))

  177: if (strategy.position_size < 0)  // If short
