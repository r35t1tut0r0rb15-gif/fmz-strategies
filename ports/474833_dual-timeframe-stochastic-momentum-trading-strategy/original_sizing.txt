## Target 2 % / stop 1 % of the signal close  [lines 111-141]
  111: tpPerc = input.float(2.0, title="Take Profit (%)", step=0.1)
  112: slPerc = input.float(1.0, title="Stop Loss (%)", step=0.1)
  113: 
  114: // Hitung Stochastic
  115: k = ta.sma(ta.stoch(close, high, low, length), smoothK)
  116: d = ta.sma(k, smoothD)
  117: 
  118: // Logika Sinyal
  119: co = ta.crossover(k, d)  // %K memotong %D ke atas
  120: cu = ta.crossunder(k, d) // %K memotong %D ke bawah
  121: 
  122: longCondition = co and k < OverSold
  123: shortCondition = cu and k > OverBought
  124: 
  125: // Harga untuk TP dan SL
  126: var float longTP = na
  127: var float longSL = na
  128: var float shortTP = na
  129: var float shortSL = na
  130: 
  131: if (longCondition)
  132:     longTP := close * (1 + tpPerc / 100)
  133:     longSL := close * (1 - slPerc / 100)
  134:     strategy.entry("Buy", strategy.long, comment="StochLE")
  135:     strategy.exit("Sell Exit", "Buy", limit=longTP, stop=longSL)
  136: 
  137: if (shortCondition)
  138:     shortTP := close * (1 - tpPerc / 100)
  139:     shortSL := close * (1 + slPerc / 100)
  140:     strategy.entry("Sell", strategy.short, comment="StochSE")
  141:     strategy.exit("Buy Exit", "Sell", limit=shortTP, stop=shortSL)
