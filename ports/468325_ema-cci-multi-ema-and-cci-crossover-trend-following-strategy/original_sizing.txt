  177: takeProfitPercent = input.float(1.5, title="Take Profit (%)", step=0.1)

  180: stopLossPercentTime = input.float(0.2, title="Stop Loss (%) for Time-based", step=0.1)
  181: max_wait_bars = input.float(2, title="Max wait candles", step=1)
  182: // Расчет CCI
  183: cci = ta.cci(close, cci_length)
  184:

  186: sz = strategy.position_size

  188: // Флаги для отслеживания пересечений EMA вверх
  189: var int ema8_cross_index_up = na
  190: var int ema12_cross_index_up = na
  191: var int ema24_cross_index_up = na
  192: 
  193: // Флаги для отслеживания пересечений EMA вниз
  194: var int ema8_cross_index_down = na
  195: var int ema12_cross_index_down = na
  196: var int ema24_cross_index_down = na
  197:

  201: if (ta.crossover(ema12, ema72))

  204:     ema24_cross_index_up := bar_index

  207: if (ta.crossunder(ema8, ema72))

  209: if (ta.crossunder(ema12, ema72))

  213: 
  214: // Условия пересечения за одну свечу (лонг и шорт)

  218: 
  219: cross_condition_one_candle_short = (na(ema8_cross_index_down) == false and (bar_index - ema8_cross_index_down) == 0) and
  220:                                    (na(ema12_cross_index_down) == false and (bar_index - ema12_cross_index_down) == 0) and
  221:                                    (na(ema24_cross_index_down) == false and (bar_index - ema24_cross_index_down) == 0)
  222: 
  223: // Условия пересечения в течение указанного времени (лонг и шорт)
  224: cross_condition_within_time_long = (not na(ema8_cross_index_up) and (bar_index - ema8_cross_index_up) <= max_wait_bars) and
  225:                                    (not na(ema12_cross_index_up) and (bar_index - ema12_cross_index_up) <= max_wait_bars) and
  226:                                    (not na(ema24_cross_index_up) and (bar_index - ema24_cross_index_up) <= max_wait_bars)
  227: 
  228: cross_condition_within_time_short = (not na(ema8_cross_index_down) and (bar_index - ema8_cross_index_down) <= max_wait_bars) and (not na(ema12_cross_index_down) and (bar_index - ema12_cross_index_down) <= max_wait_bars) and (not na(ema24_cross_index_down) and (bar_index - ema24_cross_index_down) <= max_wait_bars)
  229: 
  230: // Условие для открытия лонга
  231: long_condition_one = cross_condition_one_candle_long and cci > cci_overbought and close > ema72
  232: long_condition_time = cross_condition_within_time_long and cci > cci_overbought and close > ema72
  233: 
  234: // Условие для открытия шорта
  235: short_condition_one = cross_condition_one_candle_short and cci < cci_oversold and close < ema72
  236: short_condition_time = cross_condition_within_time_short and cci < cci_oversold and close < ema72
  237: 
  238: // Вход в лонг

  242: if (long_condition_time and sz == 0)
  243:     strategy.entry(id='Long_time', direction=strategy.long)
  244: 
  245: // Вход в шорт
  246: if (short_condition_one and sz == 0)
  247:     strategy.entry(id='Short_one', direction=strategy.short)
  248: 
  249: if (short_condition_time and sz == 0)
  250:     strategy.entry(id='Short_time', direction=strategy.short)
  251: 
  252: // Вычисление цен тейк-профита и стоп-лосса для лонга

  255:     takeProfitPriceLong = entryPriceLong * (1 + takeProfitPercent / 100)
