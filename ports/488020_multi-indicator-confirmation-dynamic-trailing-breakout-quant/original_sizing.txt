  258:

  262: signalLine_1h := signalLine_1h[1]

  264: // Trend Confirmation

  266: trendDown_1h = emaFastValue < emaSlowValue
  267: breakoutLong = ta.crossover(close, emaFastValue) and trendUp_1h and close > ta.sma(close, 20) and not na(emaFastValue)
  268: breakoutShort = ta.crossunder(close, emaSlowValue) and trendDown_1h and atrValue > ta.sma(atrValue, 14) and not na(emaSlowValue)
  269: noActivePosition = strategy.position_size == 0

  271: // === Dynamic Position Sizing ===
  272: equity = strategy.equity
  273: riskAmount = equity * (riskPct / 100)
  274: stopDistance = atrValue * 1.2  // Tightened to 1.2x ATR
  275: leverage = baseLeverage * math.min(1.0, 1.0 / (atrValue / close))
  276: positionSize = math.round((riskAmount / stopDistance) * leverage)
