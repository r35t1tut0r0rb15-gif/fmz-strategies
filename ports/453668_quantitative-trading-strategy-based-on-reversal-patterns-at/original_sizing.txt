## strategy(): 100 % of equity; stop 1 % beyond the bar's extreme, target 3 %  [lines 109-162]
  109: strategy("Reversal Patterns at Support and Resistance", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=100)
  110: 
  111: // Parameters
  112: support_resistance_lookback = input.int(50, title="Support/Resistance Lookback Period")
  113: reversal_tolerance = input.float(0.01, title="Reversal Tolerance (percent)", step=0.01) / 100
  114: take_profit_percent = input.float(3, title="Take Profit (%)") / 100
  115: stop_loss_percent = input.float(1, title="Stop Loss (%)") / 100
  116: 
  117: // Functions to identify key support and resistance levels
  118: findSupport() =>
  119:     ta.lowest(low, support_resistance_lookback)
  120: 
  121: findResistance() =>
  122:     ta.highest(high, support_resistance_lookback)
  123: 
  124: // Identify reversal patterns
  125: isHammer() =>
  126:     body = math.abs(close - open)
  127:     lowerWick = open > close ? (low < close ? close - low : open - low) : (low < open ? open - low : close - low)
  128:     upperWick = high - math.max(open, close)
  129:     lowerWick > body * 2 and upperWick < body
  130: 
  131: isEngulfing() =>
  132:     (close[1] < open[1] and close > open and close > open[1] and open < close[1]) 
  133:     (close[1] > open[1] and close < open and close < open[1] and open > close[1])
  134: 
  135: isDoji() =>
  136:     math.abs(open - close) <= (high - low) * 0.1
  137: 
  138: // Identify support and resistance levels
  139: support = findSupport()
  140: resistance = findResistance()
  141: 
  142: // Check for reversal patterns at support and resistance
  143: hammerAtSupport = isHammer() and (low <= support * (1 + reversal_tolerance))
  144: engulfingAtSupport = isEngulfing() and (low <= support * (1 + reversal_tolerance))
  145: dojiAtSupport = isDoji() and (low <= support * (1 + reversal_tolerance))
  146: 
  147: hammerAtResistance = isHammer() and (high >= resistance * (1 - reversal_tolerance))
  148: engulfingAtResistance = isEngulfing() and (high >= resistance * (1 - reversal_tolerance))
  149: dojiAtResistance = isDoji() and (high >= resistance * (1 - reversal_tolerance))
  150: 
  151: // Trading logic
  152: if (hammerAtSupport or engulfingAtSupport or dojiAtSupport)
  153:     strategy.entry("Long", strategy.long)
  154:     stop_level = low * (1 - stop_loss_percent)
  155:     take_profit_level = close * (1 + take_profit_percent)
  156:     strategy.exit("Take Profit/Stop Loss", from_entry="Long", stop=stop_level, limit=take_profit_level)
  157: 
  158: if (hammerAtResistance or engulfingAtResistance or dojiAtResistance)
  159:     strategy.entry("Short", strategy.short)
  160:     stop_level = high * (1 + stop_loss_percent)
  161:     take_profit_level = close * (1 - take_profit_percent)
  162:     strategy.exit("Take Profit/Stop Loss", from_entry="Short", stop=stop_level, limit=take_profit_level)
