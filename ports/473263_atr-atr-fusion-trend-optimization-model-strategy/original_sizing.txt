  112: tp2ATR = input.float(8.0, title="Take Profit Level 2 ATR Multiplier")
  113: tp3ATR = input.float(14.0, title="Take Profit Level 3 ATR Multiplier")

  115: // Take Profit Percentages
  116: tp1_percent = input.float(12.0, title="TP Level 1 Percentage", minval=0.0, maxval=100.0)

  118: tp3_percent = input.float(12.0, title="TP Level 3 Percentage", minval=0.0, maxval=100.0)
  119: 
  120: true_low = math.min(low, close[1])
  121: true_high = math.max(high, close[1])
  122: 
  123: // Calculate True Range
  124: true_range = true_high - true_low
  125: 
  126: // Calculate BP (Buying Pressure)
  127: bp = close - true_low
  128: 
  129: // Calculate ratios for different periods
  130: calc_ratio(len) =>
  131:     sum_bp = math.sum(bp, len)
  132:     sum_tr = math.sum(true_range, len)
  133:     100 * sum_bp / sum_tr
  134: 
  135: // Calculate weighted average of different timeframes
  136: weighted_avg = (5 * calc_ratio(8) + 4 * calc_ratio(13) + 3 * calc_ratio(21) + 2 * calc_ratio(34) + calc_ratio(55)) / (5 + 4 + 3 + 2 + 1)
  137: weighted_avg_sma = ta.sma(weighted_avg,3)
  138: 
  139: // Plot the indicator

  141: plot(weighted_avg_sma, "SMA Fibonacci ATR", color=color.yellow, linewidth=2)
  142: 
  143: // Define trading conditions
  144: longCondition = ta.crossover(weighted_avg_sma, long_entry_threshold)  // Enter long when weighted average crosses above threshold
  145: shortCondition = ta.crossunder(weighted_avg_sma, short_entry_threshold) // Enter short when weighted average crosses below threshold
  146: longExit = ta.crossunder(weighted_avg_sma, long_exit_threshold)
  147: shortExit = ta.crossover(weighted_avg_sma, short_exit_threshold)
