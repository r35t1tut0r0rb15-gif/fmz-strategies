## qty = equity x 10 % x leverage / close; percent stops / targets  [lines 196-263]
  196: leverage = input.float(1, title="Leverage", minval=1, step=0.1)
  197: 
  198: // Moving Average Settings
  199: fastLength = input(5, title="Fast Length")
  200: slowLength = input(20, title="Slow Length")
  201: 
  202: fastMA = ta.sma(close, fastLength)
  203: slowMA = ta.sma(close, slowLength)
  204: 
  205: // Trend Ribbon Settings
  206: ribbonColor = input(true, title="Show Trend Ribbon")
  207: ribbonLength = input(20, title="Ribbon Length")
  208: ribbonColorUp = color.new(color.blue, 80)
  209: ribbonColorDown = color.new(color.red, 80)
  210: 
  211: ribbonUp = ta.crossover(close, ta.sma(close, ribbonLength))
  212: ribbonDown = ta.crossunder(close, ta.sma(close, ribbonLength))
  213: 
  214: // Buy and Sell Signals
  215: buySignal = ta.crossover(close, fastMA) and ta.crossover(fastMA, slowMA)
  216: sellSignal = ta.crossunder(close, fastMA) and ta.crossunder(fastMA, slowMA)
  217: 
  218: // Input for SL/TP percentages and toggle
  219: use_sl_tp = input(true, title="Use Stop Loss/Take Profit")
  220: take_profit_long_percent = input(4.0, title="Take Profit Long (%)") / 100
  221: take_profit_short_percent = input(7.0, title="Take Profit Short (%)") / 100
  222: stop_loss_long_percent = input(2.0, title="Stop Loss Long (%)") / 100
  223: stop_loss_short_percent = input(2.0, title="Stop Loss Short (%)") / 100
  224: 
  225: // Calculate SL and TP levels
  226: calculate_sl_tp(entryPrice, isLong) =>
  227:     stopLoss = isLong ? entryPrice * (1 - stop_loss_long_percent) : entryPrice * (1 + stop_loss_short_percent)
  228:     takeProfit = isLong ? entryPrice * (1 + take_profit_long_percent) : entryPrice * (1 - take_profit_short_percent)
  229:     [stopLoss, takeProfit]
  230: 
  231: // Plotting Moving Averages
  232: plot(fastMA, color=color.blue, title="Fast MA")
  233: plot(slowMA, color=color.red, title="Slow MA")
  234: 
  235: // Plotting Trend Ribbon
  236: bgcolor(ribbonColor ? ribbonUp ? ribbonColorUp : ribbonDown ? ribbonColorDown : na : na)
  237: 
  238: // Calculate position size based on the percentage of the portfolio and leverage
  239: percentOfPortfolio = input.float(10, title="Percent of Portfolio")
  240: positionSizePercent = percentOfPortfolio / 100 * leverage
  241: positionSize = strategy.equity * positionSizePercent / close
  242: 
  243: // Strategy Execution with Leverage
  244: var float stopLossLong = na
  245: var float takeProfitLong = na
  246: var float stopLossShort = na
  247: var float takeProfitShort = na
  248: 
  249: if (buySignal)
  250:     entryPrice = close
  251:     [stopLossLong, takeProfitLong] = calculate_sl_tp(entryPrice, true)
  252:     strategy.entry("Buy", strategy.long, qty=positionSize)
  253:     if use_sl_tp
  254:         strategy.exit("Take Profit Long", "Buy", limit=takeProfitLong)
  255:         strategy.exit("Stop Loss Long", "Buy", stop=stopLossLong)
  256: 
  257: if (sellSignal)
  258:     entryPrice = close
  259:     [stopLossShort, takeProfitShort] = calculate_sl_tp(entryPrice, false)
  260:     strategy.entry("Sell", strategy.short, qty=positionSize)
  261:     if use_sl_tp
  262:         strategy.exit("Take Profit Short", "Sell", limit=takeProfitShort)
  263:         strategy.exit("Stop Loss Short", "Sell", stop=stopLossShort)
