## Entries with qty 1 and their ATR tick exits  [lines 172-191]
  172: tickFactor = isJPY? 1e3: 1e5
  173: sl = atrMulSL*atr(atrPar)*tickFactor
  174: tp = atrMulTP*atr(atrPar)*tickFactor
  175: 
  176: plot(sig1, color=aqua, title="MA1", linewidth=2)
  177: plot(sig2, color=orange, title="MA2", linewidth=2)

  179: longCondition = crossunder(sig2, sig1) and year>=yr // change the >= to == if you like exact years not a range
  180: if (longCondition)
  181:     strategy.entry("Long", strategy.long, qty=1) // exit trade when SL and TP are hit
