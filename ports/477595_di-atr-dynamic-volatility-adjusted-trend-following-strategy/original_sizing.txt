## ATR 14 x 1.5 stop / x 2.5 target  [lines 122-157]
  122: atrLength = input.int(title="ATR 長度", defval=14)
  123: atrMultiplierStop = input.float(title="ATR 停損倍數", defval=1.5)
  124: atrMultiplierTakeProfit = input.float(title="ATR 止盈倍數", defval=2.5)
  125: 
  126: // 計算 DI+ 和 DI-
  127: [diPlus, diMinus, _] = ta.dmi(diLength, adxSmoothing)
  128: 
  129: // 計算趨勢過濾均線
  130: trendFilterMA = ta.sma(close, trendFilterLength)
  131: 
  132: // 判斷趨勢方向和強度
  133: strongUpTrend = diPlus > diMinus + strengthThreshold and close > trendFilterMA
  134: strongDownTrend = diMinus > diPlus + strengthThreshold and close < trendFilterMA
  135: 
  136: // 計算 ATR
  137: atr = ta.atr(atrLength)
  138: 
  139: // 追蹤止損止盈價格 (使用 var 宣告，只在進場時更新)
  140: var float longStopPrice = na
  141: var float longTakeProfitPrice = na
  142: var float shortStopPrice = na
  143: var float shortTakeProfitPrice = na
  144: 
  145: // 進場邏輯
  146: longCondition = strongUpTrend
  147: shortCondition = strongDownTrend
  148: 
  149: if (longCondition)
  150:     strategy.entry("多單", strategy.long)
  151:     longStopPrice := close - atr * atrMultiplierStop // 進場時計算並更新止損價
  152:     longTakeProfitPrice := close + atr * atrMultiplierTakeProfit // 進場時計算並更新止盈價
  153: 
  154: if (shortCondition)
  155:     strategy.entry("空單", strategy.short)
  156:     shortStopPrice := close + atr * atrMultiplierStop // 進場時計算並更新止損價
  157:     shortTakeProfitPrice := close - atr * atrMultiplierTakeProfit // 進場時計算並更新止盈價
