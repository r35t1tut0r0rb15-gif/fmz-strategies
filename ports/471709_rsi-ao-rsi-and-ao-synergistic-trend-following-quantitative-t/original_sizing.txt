## Target 2 % / stop 1 % of the average price  [lines 111-216]
  111: takeProfitPerc = input.float(title="Take Profit (%)", defval=2.0, minval=0.0, step=0.1)
  112: stopLossPerc = input.float(title="Stop Loss (%)", defval=1.0, minval=0.0, step=0.1)

  114: // -----------------------------

  134: rsiCrossUnder50 = ta.crossunder(rsiValue, 50)

  198: // --- Strategy Execution ---
  199: // -----------------------------
  200: 
  201: // Entry Condition
  202: if buySignal

  204: 
  205: // Exit Conditions
  206: // Calculate Stop Loss and Take Profit Prices
  207: if strategy.position_size > 0
  208:     // Entry price
  209:     entryPrice = strategy.position_avg_price
  210: 
  211:     // Stop Loss and Take Profit Levels
  212:     stopLevel = entryPrice * (1 - stopLossPerc / 100)
  213:     takeProfitLevel = entryPrice * (1 + takeProfitPerc / 100)
  214: 
  215:     // Submit Stop Loss and Take Profit Orders
  216:     strategy.exit("Exit Long", from_entry="Long", stop=stopLevel, limit=takeProfitLevel)
