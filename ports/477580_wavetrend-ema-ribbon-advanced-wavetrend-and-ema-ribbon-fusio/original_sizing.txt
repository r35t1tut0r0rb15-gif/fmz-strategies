## strategy(): capital 10000, fixed qty 1; swing stop, 2 R / 3 R targets  [lines 105-208]

  107: // === 函数定义 ===
  108: // WaveTrend函数
  109: f_wavetrend(_src, _chlen, _avg, _malen) =>
  110:     _esa = ta.ema(_src, _chlen)
  111:     _de = ta.ema(math.abs(_src - _esa), _chlen)

  121:     _ema2 = ta.ema(_src, _e2)

  123:     _ema4 = ta.ema(_src, _e4)
