## Swing stops and 2 R targets  [lines 204-211]
  204: longSL = recentLow  // SL for Buy = Last 10-bar Low
  205: shortSL = recentHigh  // SL for Sell = Last 10-bar High
  206: 
  207: riskSizeLong = close - longSL  // Risk for Buy

  210: longTP = close + (riskSizeLong * 2)  // 1:2 RR TP for Buy
