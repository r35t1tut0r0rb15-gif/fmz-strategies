## Target 1000 points; stop at the trigger bar's high / low  [lines 92-175]
   92: targetPoints = 1000
   93: 
   94: // Calculate 5-day EMA
   95: ema5 = ta.ema(close, lengthEMA)

   97: // Calculate Bollinger Bands (length 20, multiplier 1.5)
   98: basis = ta.sma(close, lengthBB)
   99: dev = multBB * ta.stdev(close, lengthBB)
  100: upperBB = basis + dev
  101: lowerBB = basis - dev

  103: // Define strategy variables
  104: var float entryPrice = na
  105: var float stopLoss = na
  106: var float targetPrice = na
  107: var bool inTrade = false
  108: var bool isLong = false
  109: var float triggerHigh = na
  110: var float triggerLow = na
  111: var float triggerClose = na
  112: 
  113: if not inTrade
  114:     // Short Entry Trigger Condition
  115:     if low > ema5 and low > upperBB and high > upperBB
  116:         triggerLow := low
  117:         triggerHigh := high
  118:         triggerClose := close
  119:         label.new(bar_index, high, "Waiting for short trigger", color=color.yellow)
  120:     // Long Entry Trigger Condition
  121:     else if high < ema5 and high < lowerBB and low < lowerBB
  122:         triggerHigh := high
  123:         triggerLow := low
  124:         triggerClose := close
  125:         label.new(bar_index, low, "Waiting for long trigger", color=color.yellow)
  126: 
  127: // Check for Short Entry
  128: if not inTrade and na(triggerClose) == false and close < triggerClose
  129:     if low < triggerLow
  130:         entryPrice := close
  131:         stopLoss := triggerHigh
  132:         targetPrice := entryPrice - targetPoints
  133:         strategy.entry("Short", strategy.short)
  134:         label.new(bar_index, high, "Short", color=color.red, style=label.style_label_down)
  135:         inTrade := true
  136:         isLong := false
  137:         triggerLow := na
  138:         triggerHigh := na
  139:         triggerClose := na
  140: 
  141: // Check for Long Entry
  142: if not inTrade and na(triggerClose) == false and close > triggerClose
  143:     if high > triggerHigh
  144:         entryPrice := close
  145:         stopLoss := triggerLow
  146:         targetPrice := entryPrice + targetPoints
  147:         strategy.entry("Long", strategy.long)
  148:         label.new(bar_index, low, "Long", color=color.green, style=label.style_label_up)
  149:         inTrade := true

  151:         triggerLow := na
  152:         triggerHigh := na
  153:         triggerClose := na

  162:         strategy.close("Short", comment="Target Hit")
