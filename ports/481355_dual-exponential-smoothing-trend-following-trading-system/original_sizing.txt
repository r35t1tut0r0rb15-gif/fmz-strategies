  122: adaptivePricePosition(priceSource, lookback) =>  
  123:     weightedSum = 0.0  
  124:     smoothingCoefficient = customSmoothingFactor(lookback)  
  125:     cumulativeWeight = 0.0  
  126:     for iteration = 0 to lookback - 1 by 1  
  127:         historicalWeight = math.pow(1 - smoothingCoefficient, iteration)  
  128:         weightedSum := weightedSum + priceSource[iteration] * historicalWeight  
  129:         cumulativeWeight := cumulativeWeight + historicalWeight  
  130:     weightedSum / cumulativeWeight
