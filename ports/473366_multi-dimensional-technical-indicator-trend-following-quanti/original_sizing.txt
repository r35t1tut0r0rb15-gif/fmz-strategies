## Capital 100, risk 1 %, TP 2 %, SL 1 % (labels only); stop close x 0.99 after 5 % profit  [lines 107-142]

  122: ema_50 = ta.ema(close, 50)

  140: // Adiciona trailing stop automático
  141: if (strategy.position_size > 0 and close >= strategy.position_avg_price * (1 + trailing_stop_percent / 100))
