   94: // Параметры стратегии

  100: calcRiskRewardPrice(entryPrice, stopLossPrice, riskRewardRatio) =>
  101:     takeProfitPrice = entryPrice + (entryPrice - stopLossPrice) * riskRewardRatio
  102:     [stopLossPrice, takeProfitPrice]
  103: 
  104: // Найти минимум и максимум флага
  105: flagLow = ta.lowest(low, flagLength)
  106: flagHigh = ta.highest(high, flagLength)
  107: 
  108: // Условия для формирования бычьего флага
  109: isBullFlag = high[1] < flagHigh and close > high[1]
  110: 
  111: // Условия для входа в сделку
  112: if (isBullFlag)
  113:     entryPrice = close
  114:     stopLossPrice = flagLow - stopLossBuffer
  115:     [calculatedStopLoss, calculatedTakeProfit] = calcRiskRewardPrice(entryPrice, stopLossPrice, riskRewardRatio)
  116:     
  117:     // Открыть длинную позицию
  118:     strategy.entry("Bull Flag Long", strategy.long)
  119:     strategy.exit("Take Profit", "Bull Flag Long", limit=calculatedTakeProfit)
  120:     strategy.exit("Stop Loss", "Bull Flag Long", stop=calculatedStopLoss)
