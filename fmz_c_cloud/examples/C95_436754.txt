import numpy as np
import pandas as pd

NAME = "fmz_436754_rsi_threshold_alternating_bracket"
FAMILY = "rsi_oscillator"  # proposed 2026-10-21 (FMZ-C), user to confirm
FREQ = "3min"  # backtest header period: 3m
PERIODS_PER_YEAR_OVERRIDE = None
USES_STOPS = True

GRID = {
    "period": [14, 21],
    "k_atr": [0.6, 1.5],
}
DEFAULT_PARAMS = {"period": 14, "k_atr": 0.6, "up": 75.0, "dn": 25.0, "inverse": False, "tp_ratio": 10.0,
                  "atr_len": 14}


def _resample(raw_1m_df, freq):
    ohlc = raw_1m_df[["open", "high", "low", "close"]]
    bars = ohlc.resample(freq, label="left", closed="left").agg(
        {"open": "first", "high": "max", "low": "min", "close": "last"})
    bars = bars.dropna(how="all")  # an empty period is not a bar; never forward-filled
    if bars.index.tz is None:
        raise ValueError("raw_1m_df must have a tz-aware UTC index")
    return bars


def _rma(x, n):
    """Wilder smoothing as Pine's rma: SMA seed over the first n valid values, then recursive."""
    v = x.to_numpy(dtype=float)
    out = np.full(v.shape, np.nan)
    valid = np.flatnonzero(~np.isnan(v))
    if len(valid) >= n:
        s = valid[0]
        out[s + n - 1] = v[s:s + n].mean()
        for i in range(s + n, len(v)):
            out[i] = (out[i - 1] * (n - 1) + v[i]) / n
    return pd.Series(out, index=x.index)


def _rsi(c, n):
    d = c.diff()
    up = _rma(d.clip(lower=0), n)
    dn = _rma((-d).clip(lower=0), n)
    with np.errstate(divide="ignore", invalid="ignore"):
        r = 100.0 - 100.0 / (1.0 + up / dn)
    r = r.where(up != 0, 0.0).where(dn != 0, 100.0)    # Pine: down == 0 -> 100, else up == 0 -> 0
    return r.where(up.notna() & dn.notna())


def _atr(bars_df, n):
    h, lo, c = bars_df["high"], bars_df["low"], bars_df["close"]
    pc = c.shift(1)
    tr = pd.concat([h - lo, (h - pc).abs(), (lo - pc).abs()], axis=1).max(axis=1)
    return _rma(tr, int(n))


def precompute(raw_1m_df, symbol_key, **params):
    return _resample(raw_1m_df, FREQ)


def simulate(bars_df, **params):
    p = {**DEFAULT_PARAMS, **params}
    r = _rsi(bars_df["close"], int(p["period"]))
    r1 = r.shift(1)
    rising, falling = r > r1, r < r1

    def cross(t):
        return ((r > t) & (r1 <= t)) | ((r < t) & (r1 >= t))

    up_x, dn_x = cross(float(p["up"])), cross(float(p["dn"]))
    if p["inverse"]:
        buy, sell = falling & dn_x, rising & up_x
    else:
        buy, sell = rising & up_x, falling & dn_x
    b, s = buy.to_numpy(), sell.to_numpy()
    m = len(b)
    le, se = np.zeros(m, dtype=bool), np.zeros(m, dtype=bool)
    prev = None                                  # myPosition[1]; na on the first bar
    for i in range(m):
        if b[i] and prev == 1:
            le[i] = True
        if s[i] and prev == 0:
            se[i] = True
        prev = 0 if b[i] else (1 if (s[i] or prev == 1) else 0)
    ix = bars_df.index
    false = pd.Series(False, index=ix)
    return pd.Series(le, index=ix), false, pd.Series(se, index=ix), false.copy()


def stops(bars_df, **params):
    p = {**DEFAULT_PARAMS, **params}
    f = float(p["k_atr"]) * _atr(bars_df, p["atr_len"]) / bars_df["close"]
    return {"sl_stop": f.shift(1), "tp_stop": (float(p["tp_ratio"]) * f).shift(1)}

    return {}
